## 2.4.1 SQLite 数据库结构

### 数据库表设计

```sql
-- memory/database.py

import sqlite3
from pathlib import Path

DB_PATH = Path("/opt/cybercat/memory.db")

def init_database():
    """初始化数据库，创建表结构"""
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()
    
    # 会话表：存储短期对话历史
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS sessions (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            session_id TEXT NOT NULL,
            role TEXT NOT NULL,          -- 'user' 或 'assistant'
            content TEXT NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)
    
    # 长期记忆表：存储用户偏好和重要信息
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS long_term_memory (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            session_id TEXT NOT NULL,
            summary TEXT NOT NULL,
            updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            UNIQUE(session_id)
        )
    """)
    
    # 设备表：注册设备信息
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS devices (
            device_id TEXT PRIMARY KEY,
            name TEXT,
            location TEXT,
            last_seen TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)
    
    # 创建索引（加速查询）
    cursor.execute("""
        CREATE INDEX IF NOT EXISTS idx_sessions_session_id 
        ON sessions(session_id, created_at)
    """)
    
    conn.commit()
    conn.close()
    print("[Database] 初始化完成")

if __name__ == "__main__":
    init_database()
```

### 表结构说明

**sessions 表（短期记忆）：**
| 字段 | 类型 | 说明 |
|------|------|------|
| id | INTEGER | 主键，自增 |
| session_id | TEXT | 会话 ID（设备 ID）|
| role | TEXT | 'user' 或 'assistant' |
| content | TEXT | 对话内容 |
| created_at | TIMESTAMP | 创建时间 |

**long_term_memory 表（长期记忆）：**
| 字段 | 类型 | 说明 |
|------|------|------|
| id | INTEGER | 主键，自增 |
| session_id | TEXT | 会话 ID（外键）|
| summary | TEXT | 记忆摘要 |
| updated_at | TIMESTAMP | 更新时间 |

**devices 表（设备信息）：**
| 字段 | 类型 | 说明 |
|------|------|------|
| device_id | TEXT | 主键 |
| name | TEXT | 设备名称 |
| location | TEXT | 位置 |
| last_seen | TIMESTAMP | 最后在线时间 |

### 数据库操作封装

```python
# memory/repository.py

import sqlite3
from datetime import datetime
from typing import List, Dict, Optional

DB_PATH = "/opt/cybercat/memory.db"

def get_connection():
    """获取数据库连接"""
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row  # 返回字典格式
    return conn

def save_interaction(session_id: str, user_text: str, assistant_text: str):
    """保存一轮对话到短期记忆"""
    conn = get_connection()
    cursor = conn.cursor()
    
    # 保存用户消息
    cursor.execute(
        "INSERT INTO sessions (session_id, role, content) VALUES (?, ?, ?)",
        (session_id, "user", user_text)
    )
    
    # 保存助手回复
    cursor.execute(
        "INSERT INTO sessions (session_id, role, content) VALUES (?, ?, ?)",
        (session_id, "assistant", assistant_text)
    )
    
    conn.commit()
    conn.close()

def get_session_history(session_id: str, limit: int = 10) -> str:
    """获取最近对话历史"""
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute("""
        SELECT role, content, created_at 
        FROM sessions 
        WHERE session_id = ?
        ORDER BY created_at DESC
        LIMIT ?
    """, (session_id, limit))
    
    rows = cursor.fetchall()
    conn.close()
    
    # 反转顺序（从旧到新）
    rows.reverse()
    
    # 格式化为字符串
    history = []
    for row in rows:
        role_label = "你" if row["role"] == "user" else "猫"
        history.append(f"[{role_label}] {row['content']}")
    
    return "\n".join(history)

def get_memory_summary(session_id: str) -> Optional[str]:
    """获取长期记忆摘要"""
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute(
        "SELECT summary FROM long_term_memory WHERE session_id = ?",
        (session_id,)
    )
    
    row = cursor.fetchone()
    conn.close()
    
    return row["summary"] if row else None

def update_memory_summary(session_id: str, summary: str):
    """更新长期记忆摘要"""
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute("""
        INSERT OR REPLACE INTO long_term_memory (session_id, summary, updated_at)
        VALUES (?, ?, CURRENT_TIMESTAMP)
    """, (session_id, summary))
    
    conn.commit()
    conn.close()

def register_device(device_id: str, name: str = None, location: str = None):
    """注册设备信息"""
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute("""
        INSERT OR REPLACE INTO devices (device_id, name, location, last_seen)
        VALUES (?, ?, ?, CURRENT_TIMESTAMP)
    """, (device_id, name, location))
    
    conn.commit()
    conn.close()

def get_all_devices() -> List[Dict]:
    """获取所有设备列表"""
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute("SELECT * FROM devices ORDER BY last_seen DESC")
    
    devices = [dict(row) for row in cursor.fetchall()]
    conn.close()
    
    return devices
```

## 2.4.2 短期记忆 (Sessions)

### 短期记忆的特点

- **时效性**：保留最近 N 轮对话
- **完整性**：保存完整的对话内容
- **按会话隔离**：每个设备有独立的短期记忆

### 实现逻辑

```python
# memory/short_term.py

from memory.repository import get_connection

MAX_SHORT_TERM_ROUNDS = 10  # 保留最近 10 轮

def get_short_term_memory(session_id: str) -> List[Dict]:
    """
    获取短期记忆（最近 N 轮对话）
    返回：[{"role": "user", "content": "..."}, ...]
    """
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute("""
        SELECT role, content 
        FROM sessions 
        WHERE session_id = ?
        ORDER BY created_at DESC
        LIMIT ?
    """, (session_id, MAX_SHORT_TERM_ROUNDS * 2))  # 每轮包含 user+assistant
    
    rows = cursor.fetchall()
    conn.close()
    
    # 反转并从 Dict 转为标准格式
    return [
        {"role": row["role"], "content": row["content"]}
        for row in reversed(rows)
    ]

def format_short_term_for_prompt(session_id: str) -> str:
    """
    格式化为 LLM 可读的提示词
    """
    memories = get_short_term_memory(session_id)
    
    if not memories:
        return "[暂无对话历史]"
    
    lines = []
    for m in memories:
        role_label = "你" if m["role"] == "user" else "猫"
        lines.append(f"[{role_label}] {m['content']}")
    
    return "\n".join(lines)
```

### 清理过期数据

```python
# memory/cleanup.py

from memory.repository import get_connection

CLEANUP_THRESHOLD_DAYS = 7  # 清理 7 天前的数据

def cleanup_old_sessions():
    """清理过期的会话数据"""
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute("""
        DELETE FROM sessions 
        WHERE created_at < datetime('now', '-7 days')
    """)
    
    deleted = cursor.rowcount
    
    conn.commit()
    conn.close()
    
    print(f"[Cleanup] 删除了 {deleted} 条过期会话记录")
    return deleted

def cleanup_old_sessions_by_count(session_id: str, keep_count: int = 50):
    """按数量清理，保留最近 N 条"""
    conn = get_connection()
    cursor = conn.cursor()
    
    # 删除超出保留数量的记录
    cursor.execute("""
        DELETE FROM sessions 
        WHERE session_id = ? AND id NOT IN (
            SELECT id FROM sessions 
            WHERE session_id = ?
            ORDER BY created_at DESC
            LIMIT ?
        )
    """, (session_id, session_id, keep_count))
    
    deleted = cursor.rowcount
    conn.commit()
    conn.close()
    
    return deleted
```

## 2.4.3 长期记忆 (摘要)

### 长期记忆的特点

- **持久性**：永久保存，除非手动删除
- **摘要性**：提炼关键信息，不是完整对话
- **可更新**：新信息会合并到现有记忆中

### 记忆摘要生成

```python
# memory/long_term.py

from services import llm
from memory.repository import get_connection, get_session_history

SUMMARY_GENERATION_PROMPT = """
请从以下对话中提取关于用户的关键信息，用简洁的语句总结。
包括：用户的名字、喜好、重要事件、习惯等。
格式要求：简短的条目式总结，每条不超过 20 字。

对话历史：
{history}

记忆摘要：
"""

def generate_memory_summary(session_id: str) -> str:
    """
    基于对话历史生成记忆摘要
    """
    # 获取最近对话
    history = get_session_history(session_id, limit=20)
    
    # 调用 LLM 生成摘要
    prompt = SUMMARY_GENERATION_PROMPT.format(history=history)
    summary = llm.generate_completion(prompt, max_tokens=200)
    
    return summary.strip()

def should_update_memory(session_id: str) -> bool:
    """
    判断是否需要更新记忆
    策略：每 5 轮对话更新一次
    """
    conn = get_connection()
    cursor = conn.cursor()
    
    # 获取当前对话数
    cursor.execute("""
        SELECT COUNT(*) as cnt FROM sessions 
        WHERE session_id = ?
    """, (session_id,))
    
    count = cursor.fetchone()["cnt"]
    conn.close()
    
    # 每 5 轮更新一次
    return count > 0 and count % 5 == 0

def update_memory_if_needed(session_id: str) -> bool:
    """
    检查并更新记忆
    返回：是否执行了更新
    """
    if not should_update_memory(session_id):
        return False
    
    summary = generate_memory_summary(session_id)
    update_memory_summary(session_id, summary)
    
    print(f"[Memory] 更新会话 {session_id} 的记忆：{summary}")
    return True
```

### 记忆合并策略

```python
def merge_memory(session_id: str, new_info: str):
    """
    将新信息合并到现有记忆中
    
    策略：
    1. 获取现有记忆
    2. 用 LLM 合并新旧信息
    3. 去重、解决冲突
    """
    existing = get_memory_summary(session_id) or ""
    
    if not existing:
        # 没有现有记忆，直接保存
        update_memory_summary(session_id, new_info)
        return
    
    merge_prompt = f"""
现有记忆：
{existing}

新信息：
{new_info}

请合并以上信息，去重并解决可能的冲突。
保持简洁的条目式格式。

合并后的记忆：
"""
    
    merged = llm.generate_completion(merge_prompt, max_tokens=300)
    update_memory_summary(session_id, merged.strip())
```

## 2.4.4 记忆检索与注入

### 检索策略

```python
# memory/retrieval.py

from memory.repository import get_connection

def search_memories(session_id: str, query: str, top_k: int = 3) -> List[Dict]:
    """
    搜索相关记忆（简单版本：关键词匹配）
    
    高级版本可用向量数据库（如 Chroma、Pinecone）
    """
    conn = get_connection()
    cursor = conn.cursor()
    
    # 简单关键词匹配
    cursor.execute("""
        SELECT content, created_at 
        FROM sessions 
        WHERE session_id = ? AND content LIKE ?
        ORDER BY created_at DESC
        LIMIT ?
    """, (session_id, f"%{query}%", top_k))
    
    results = [dict(row) for row in cursor.fetchall()]
    conn.close()
    
    return results

def get_context_memories(session_id: str, current_topic: str) -> str:
    """
    获取与当前话题相关的背景记忆
    """
    # 提取话题关键词（简单实现：直接用输入）
    related = search_memories(session_id, current_topic, top_k=3)
    
    if not related:
        return ""
    
    lines = ["[相关记忆]"]
    for r in related:
        lines.append(f"- {r['content']}")
    
    return "\n".join(lines)
```

### 注入 LLM 提示词

```python
# services/llm.py

from memory.repository import get_memory_summary, get_session_history

SYSTEM_PROMPT = """你是一只可爱的电子猫咪，名叫"赛博喵"。
用简短的一两句话回复，可以加上喵~等拟声词。
你的性格：好奇、友好、偶尔撒娇。
"""

async def chat_completion(
    user_input: str,
    session_id: str = "default",
    memory_summary: str = None,
    session_history: str = None
) -> str:
    """
    LLM 聊天，注入记忆
    """
    # 获取记忆
    if memory_summary is None:
        memory_summary = get_memory_summary(session_id) or ""
    
    if session_history is None:
        session_history = get_session_history(session_id, limit=10)
    
    # 构建系统提示词
    system_content = SYSTEM_PROMPT
    
    if memory_summary:
        system_content += f"\n\n[关于用户的记忆]\n{memory_summary}"
    
    # 构建消息
    messages = [{"role": "system", "content": system_content}]
    
    # 注入对话历史
    if session_history:
        messages.append({
            "role": "user", 
            "content": f"[最近对话]\n{session_history}"
        })
    
    # 用户当前输入
    messages.append({"role": "user", "content": user_input})
    
    # 调用 API
    async with httpx.AsyncClient() as client:
        response = await client.post(
            DASHSCOPE_API_BASE,
            headers={"Authorization": f"Bearer {DASHSCOPE_KEY}"},
            json={"model": LLM_MODEL, "messages": messages},
            timeout=30
        )
        
        response.raise_for_status()
        return response.json()["choices"][0]["message"]["content"]
```

### 完整流程

```python
# main.py

from fastapi import FastAPI, Request, Header
from services import stt, llm, tts, emotion
from memory import repository, long_term

app = FastAPI()

@app.post("/chat")
async def chat(
    request: Request,
    x_session_id: str | None = Header(None)
):
    # 1. 读取音频
    wav_bytes = await request.body()
    
    # 2. STT
    transcript = await stt.transcribe(wav_bytes)
    
    # 3. 获取会话 ID
    session_id = x_session_id or "default"
    
    # 4. 获取记忆
    memory_summary = repository.get_memory_summary(session_id)
    session_history = repository.get_session_history(session_id)
    
    # 5. LLM（注入记忆）
    reply = await llm.chat_completion(
        transcript,
        session_id=session_id,
        memory_summary=memory_summary,
        session_history=session_history
    )
    
    # 6. 情绪分析
    emo = emotion.analyze(reply)
    
    # 7. TTS
    import uuid
    audio_filename = f"{uuid.uuid4().hex}.wav"
    audio_url = await tts.generate_speech(reply, audio_filename)
    
    # 8. 保存对话
    repository.save_interaction(session_id, transcript, reply)
    
    # 9. 检查是否需要更新长期记忆
    if long_term.should_update_memory(session_id):
        summary = long_term.generate_memory_summary(session_id)
        repository.update_memory_summary(session_id, summary)
    
    return {
        "text_reply": reply,
        "emotion": emo,
        "audio_url": audio_url
    }
```

## 常见问题

**Q: 数据库文件在哪里？**  
A: 默认在 `/opt/cybercat/memory.db`，可用 `.schema` 查看结构。

**Q: 记忆摘要生成太慢？**  
A: 可用后台任务异步生成，或降低调用频率（如每 10 轮更新一次）。

**Q: 多个设备如何隔离记忆？**  
A: 用 `session_id`（设备 ID）作为隔离键，每个设备独立存储。
