## 2.3.1 ESP32↔后端 HTTP 合约

### 接口总览

| 端点 | 方法 | 请求体 | 响应 | 用途 |
|------|------|--------|------|------|
| `/chat` | POST | audio/wav | JSON | 主聊天接口 |
| `/health` | GET | - | JSON | 健康检查 |
| `/tts/{file}` | GET | - | audio/wav | TTS 音频下载 |

### /chat 接口详解

**请求格式：**

```http
POST /chat HTTP/1.1
Host: 122.51.129.89:8000
Content-Type: audio/wav
Content-Length: 143084
X-Session-ID: device-001
X-Device-Name: 小喵一号

[44 字节 WAV 头 + PCM 数据]
```

**响应格式：**

```http
HTTP/1.1 200 OK
Content-Type: application/json

{
    "text_reply": "喵~听到了听到了！有什么想聊的吗？",
    "emotion": "happy",
    "audio_url": "http://122.51.129.89:8000/tts/b9544fd93a674e1fa351f2e113b7481b.wav"
}
```

**字段说明：**

| 字段 | 类型 | 必填 | 说明 |
|------|------|------|------|
| `text_reply` | string | 是 | AI 回复的文本 |
| `emotion` | string | 是 | 情绪标签：neutral/happy/thinking/talking |
| `audio_url` | string | 是 | TTS 音频下载链接 |

// cloud_client.cpp

#include "cloud_client.h"
#include "cat_face.h"
#include <WiFiClientSecure.h>
#include <ArduinoJson.h>

CloudClient::CloudClient() 
    : _state(CloudState::IDLE), _responseBuffer(nullptr) {
}

CloudClient::~CloudClient() {
    if (_responseBuffer) free(_responseBuffer);
}

void CloudClient::begin(CloudCallback cb) {
    _callback = cb;
}

bool CloudClient::processChat(int16_t* pcmData, size_t sampleCount) {
    if (_state != CloudState::IDLE) {
        _lastError = "云客户端忙";
        return false;
    }
    
    _state = CloudState::UPLOADING;
    if (_callback) _callback(_state, "上传音频...");
    
    // 构建 WAV 数据
    uint32_t pcmSize = sampleCount * sizeof(int16_t);
    uint32_t wavSize = 44 + pcmSize;
    
    // 分配 PSRAM 缓冲区
    uint8_t* wavBuffer = (uint8_t*)ps_malloc(wavSize);
    if (!wavBuffer) {
        _state = CloudState::ERROR;
        _lastError = "PSRAM 不足";
        return false;
    }
    
    // 填充 WAV 头
    struct WavHeader {
        char riff[4] = "RIFF";
        uint32_t riffSize;
        char wave[4] = "WAVE";
        char fmt[4] = "fmt ";
        uint32_t fmtSize = 16;
        uint16_t audioFormat = 1;
        uint16_t numChannels = 1;
        uint32_t sampleRate = 16000;
        uint32_t byteRate = 32000;
        uint16_t blockAlign = 2;
        uint16_t bitsPerSample = 16;
        char data[4] = "data";
        uint32_t dataSize;
    };
    
    WavHeader* header = (WavHeader*)wavBuffer;
    header->riffSize = pcmSize + 36;
    header->dataSize = pcmSize;
    memcpy(wavBuffer + 44, pcmData, pcmSize);
    
    // HTTP 连接
    WiFiClientSecure client;
    client.setInsecure();
    
    if (!client.connect(BACKEND_HOST, BACKEND_PORT)) {
        _state = CloudState::ERROR;
        _lastError = "无法连接后端";
        free(wavBuffer);
        return false;
    }
    
    // 发送请求
    client.print("POST /chat HTTP/1.1\r\n");
    client.print("Host: " + String(BACKEND_HOST) + "\r\n");
    client.print("Content-Type: audio/wav\r\n");
    client.print("Content-Length: " + String(wavSize) + "\r\n");
    client.print("X-Session-ID: " + String(DEVICE_ID) + "\r\n");
    client.print("\r\n");
    client.write(wavBuffer, wavSize);
    
    free(wavBuffer);
    
    // 等待响应
    uint32_t timeout = millis() + 30000;
    while (!client.available() && millis() < timeout) {
        delay(10);
        yield();
    }
    
    if (!client.available()) {
        _state = CloudState::ERROR;
        _lastError = "响应超时";
        return false;
    }
    
    // 读取状态行
    String statusLine = client.readStringUntil('\n');
    Serial.println("[Cloud] 响应：" + statusLine);
    
    // 检查状态码
    if (!statusLine.startsWith("HTTP/1.1 200")) {
        _state = CloudState::ERROR;
        _lastError = "HTTP 错误：" + statusLine;
        return false;
    }
    
    // 跳过响应头
    while (client.available()) {
        String line = client.readStringUntil('\n');
        if (line == "\r" || line.length() == 0) break;
    }
    
    // 读取响应体（JSON）
    String jsonBody = client.readString();
    Serial.println("[Cloud] JSON: " + jsonBody);
    
    // 解析 JSON
    DynamicJsonDocument doc(2048);
    DeserializationError err = deserializeJson(doc, jsonBody);
    
    if (err) {
        _state = CloudState::ERROR;
        _lastError = "JSON 解析失败";
        return false;
    }
    
    const char* reply = doc["text_reply"];
    const char* emotion = doc["emotion"];
    const char* audioUrl = doc["audio_url"];
    
    if (!reply || !emotion || !audioUrl) {
        _state = CloudState::ERROR;
        _lastError = "JSON 字段缺失";
        return false;
    }
    
    // 设置表情
    CatExpression expr = CAT_NEUTRAL;
    if (strcmp(emotion, "happy") == 0) expr = CAT_HAPPY;
    else if (strcmp(emotion, "thinking") == 0) expr = CAT_THINKING;
    else if (strcmp(emotion, "talking") == 0) expr = CAT_TALKING;
    
    // 下载 TTS 音频
    _state = CloudState::DOWNLOADING;
    if (_callback) _callback(_state, String(audioUrl));
    
    bool success = downloadAndPlayTTS(audioUrl);
    
    _state = success ? CloudState::COMPLETE : CloudState::ERROR;
    return success;
}

bool CloudClient::downloadAndPlayTTS(const char* url) {
    // 解析 URL
    String urlStr(url);
    int hostStart = urlStr.indexOf("//") + 2;
    int hostEnd = urlStr.indexOf("/", hostStart);
    String host = urlStr.substring(hostStart, hostEnd);
    String path = urlStr.substring(hostEnd);
    
    WiFiClientSecure client;
    client.setInsecure();
    
    if (!client.connect(host.c_str(), 8000)) {
        _lastError = "无法连接 TTS 服务器";
        return false;
    }
    
    // 发送 GET 请求
    client.print("GET " + path + " HTTP/1.1\r\n");
    client.print("Host: " + host + "\r\n");
    client.print("\r\n");
    
    // 等待响应头
    uint32_t timeout = millis() + 10000;
    while (!client.available() && millis() < timeout) {
        delay(10);
    }
    
    // 读取 Content-Length
    int contentLength = 0;
    while (client.available()) {
        String line = client.readStringUntil('\n');
        if (line.startsWith("Content-Length:")) {
            contentLength = line.substring(15).toInt();
        }
        if (line == "\r" || line.length() == 0) break;
    }
    
    if (contentLength == 0) {
        _lastError = "无效的 Content-Length";
        return false;
    }
    
    // 分配缓冲区
    uint8_t* ttsBuffer = (uint8_t*)ps_malloc(contentLength);
    if (!ttsBuffer) {
        _lastError = "PSRAM 不足";
        return false;
    }
    
    // 读取音频数据
    size_t bytesRead = 0;
    while (bytesRead < contentLength) {
        size_t toRead = min((size_t)1024, contentLength - bytesRead);
        int read = client.read(ttsBuffer + bytesRead, toRead);
        if (read > 0) {
            bytesRead += read;
        } else {
            break;
        }
    }
    
    Serial.printf("[Cloud] TTS 下载完成：%d bytes\n", bytesRead);
    
    // 播放音频（调用 es8311_spk.cpp）
    extern void es8311Play(uint8_t* wavData, size_t size);
    es8311Play(ttsBuffer, bytesRead);
    
    free(ttsBuffer);
    return true;
}

### 错误处理

```cpp
// 错误码定义
enum CloudError {
    ERR_NONE = 0,
    ERR_CONNECTION_FAILED,    // 无法连接后端
    ERR_TIMEOUT,              // 响应超时
    ERR_HTTP_ERROR,           // HTTP 状态码非 200
    ERR_JSON_PARSE,           // JSON 解析失败
    ERR_FIELD_MISSING,        // JSON 字段缺失
    ERR_PSRAM_EXHAUSTED,      // PSRAM 不足
    ERR_TTS_DOWNLOAD_FAILED,  // TTS 下载失败
    ERR_AUDIO_PLAY_FAILED,    // 音频播放失败
};

// 错误信息表
const char* getErrorMessage(CloudError err) {
    switch (err) {
        case ERR_CONNECTION_FAILED: return "无法连接后端";
        case ERR_TIMEOUT: return "响应超时";
        case ERR_HTTP_ERROR: return "HTTP 错误";
        case ERR_JSON_PARSE: return "JSON 解析失败";
        case ERR_FIELD_MISSING: return "JSON 字段缺失";
        case ERR_PSRAM_EXHAUSTED: return "PSRAM 不足";
        case ERR_TTS_DOWNLOAD_FAILED: return "TTS 下载失败";
        case ERR_AUDIO_PLAY_FAILED: return "音频播放失败";
        default: return "未知错误";
    }
}

# services/stt.py

import httpx

SILICONFLOW_KEY = "sk-xxx"
STT_API_URL = "https://api.siliconflow.cn/v1/audio/transcriptions"
STT_MODEL = "FunAudioLLM/SenseVoiceSmall"

async def transcribe(wav_bytes: bytes) -> str:
    """
    SiliconFlow STT API
    文档：https://docs.siliconflow.cn/cn/api-reference/audio/audio-transcriptions
    """
    async with httpx.AsyncClient() as client:
        response = await client.post(
            STT_API_URL,
            headers={"Authorization": f"Bearer {SILICONFLOW_KEY}"},
            files={"file": ("audio.wav", wav_bytes, "audio/wav")},
            data={"model": STT_MODEL},
            timeout=30
        )
        
        response.raise_for_status()
        result = response.json()
        
        # 返回格式：{"text": "转写结果"}
        return result.get("text", "")
```

### 设备端重试机制

```cpp
// cloud_client.cpp - 带重试的上传

#define MAX_RETRIES 3
#define RETRY_DELAY_MS 2000

bool CloudClient::processChatWithRetry(int16_t* pcmData, size_t sampleCount) {
    for (int attempt = 1; attempt <= MAX_RETRIES; attempt++) {
        Serial.printf("[Cloud] 尝试 %d/%d\n", attempt, MAX_RETRIES);
        
        bool success = processChat(pcmData, sampleCount);
        if (success) {
            return true;
        }
        
        Serial.println("[Cloud] 失败：" + _lastError);
        
        if (attempt < MAX_RETRIES) {
            Serial.printf("[Cloud] %dms 后重试...\n", RETRY_DELAY_MS);
            delay(RETRY_DELAY_MS);
        }
    }
    
    _state = CloudState::ERROR;
    _lastError = "重试 " + String(MAX_RETRIES) + " 次后仍失败";
    return false;
}
```

### 后端重试机制

```python
# services/retry.py

import asyncio
import httpx
from typing import Callable, Any

class RetryConfig:
    MAX_RETRIES = 3
    INITIAL_DELAY = 1.0  # 秒
    MAX_DELAY = 10.0
    EXPONENTIAL_BASE = 2

async def retry_async(
    func: Callable,
    *args,
    exceptions: tuple = (httpx.RequestError,),
    **kwargs
) -> Any:
    """
    异步函数重试装饰器
    
    使用示例：
    result = await retry_async(
        stt.transcribe,
        wav_bytes,
        exceptions=(httpx.TimeoutException, httpx.NetworkError)
    )
    """
    last_exception = None
    delay = RetryConfig.INITIAL_DELAY
    
    for attempt in range(RetryConfig.MAX_RETRIES):
        try:
            return await func(*args, **kwargs)
        except exceptions as e:
            last_exception = e
            if attempt < RetryConfig.MAX_RETRIES - 1:
                print(f"[Retry] 失败，{delay}s 后重试... ({e})")
                await asyncio.sleep(delay)
                delay = min(delay * RetryConfig.EXPONENTIAL_BASE, RetryConfig.MAX_DELAY)
            else:
                print(f"[Retry] 重试 {RetryConfig.MAX_RETRIES} 次后放弃")
    
    raise last_exception
```

## 常见问题

**Q: 后端 API 调用失败如何调试？**  
A: 先直接用 curl/Postman 测试 API，确认 Key 有效。然后检查后端日志。

**Q: 设备端 PSRAM 不足怎么办？**  
A: 减小缓冲区大小，或用分块传输。检查是否有内存泄漏。

**Q: HTTPS 证书验证失败？**  
A: 开发环境可用 `setInsecure()`，生产环境需配置根证书。
