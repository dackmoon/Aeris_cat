## 2.2.1 音频数据采集

### 采集链路总览

```
声波 → 麦克风 → ES7210 ADC → I2S → ESP32 DMA → PSRAM 缓冲区 → VAD 处理
```

**硬件链路：**
```
双模拟麦克风 (SPH0641LM4H 等效)
    │
    │ 模拟信号 (左右声道)
    ↓
ES7210 音频 ADC
    │ - 采样率：16kHz
    │ - 位深：16-bit
    │ - 增益：24dB (可调)
    │ I2S 数据输出
    ↓
ESP32-S3 I2S 接口
    │ - DIN: GPIO39
    │ - BCK: GPIO48
    │ - LRCK: GPIO38
    │ - MCLK: GPIO2
    │ DMA 传输
    ↓
PSRAM 环形缓冲区
    │ - 最大 30 秒 (960KB)
    │ - 环形写入
    ↓
VAD 检测
    │ - RMS 能量计算
    │ - 静音超时判断
    ↓
有效语音数据
```

### ES7210 初始化序列

```cpp
// ES7210 寄存器配置（V2 板）
const uint8_t es7210Init[][2] = {
    {0x00, 0x41},  // 软件复位
    {0x01, 0x41},  // 时钟配置：MCLK 使能，I2S 模式
    {0x02, 0x12},  // ADC 配置：16kHz, 16-bit
    {0x03, 0x03},  // 数字音量：0dB
    {0x04, 0x1C},  // ADC 混合器配置
    {0x05, 0x18},  // 模拟增益：24dB（关键！太高会削波）
    {0x06, 0x00},  // 保留
    {0x07, 0x01},  // 三态门控制
    {0xFF, 0xFF}   // 结束标记
};

void es7210Init() {
    Wire.beginTransmission(0x40);  // ES7210 I2C 地址
    for (int i = 0; es7210Init[i][0] != 0xFF; i++) {
        Wire.write(es7210Init[i][0]);
        Wire.write(es7210Init[i][1]);
    }
    Wire.endTransmission();
    
    // 等待 ADC 稳定
    delay(50);
}
```

### 录音流程

```cpp
// 录音主循环（FreeRTOS 任务）
void audioRecordTask(void* parameter) {
    int16_t dmaBuffer[256];  // DMA 缓冲区大小
    size_t bytesRead;
    
    while (true) {
        if (isRecording) {
            // 从 I2S 读取数据
            i2s_read(I2S_NUM_0, dmaBuffer, sizeof(dmaBuffer), 
                     &bytesRead, portMAX_DELAY);
            
            // 处理样本（提取左声道）
            int sampleCount = bytesRead / sizeof(int16_t);
            processSamples(dmaBuffer, sampleCount);
        } else {
            vTaskDelay(10 / portTICK_PERIOD_MS);
        }
    }
}

// 样本处理（VAD 检测）
void processSamples(int16_t* samples, int count) {
    float rms = calculateRMS(samples, count);
    
    if (rms > VAD_THRESHOLD) {
        // 检测到语音
        if (!isSpeechActive) {
            isSpeechActive = true;
            speechStartMs = millis();
            silenceStartMs = 0;
        }
        silenceStartMs = 0;  // 重置静音计时
    } else {
        // 静音
        if (isSpeechActive && silenceStartMs == 0) {
            silenceStartMs = millis();
        }
        
        // 检查静音超时
        if (isSpeechActive && 
            millis() - silenceStartMs > VAD_SILENCE_MS) {
            isSpeechActive = false;
            stopRecording();  // 录音结束，触发 STT
        }
    }
    
    // 存储到环形缓冲区
    if (isSpeechActive) {
        ringBufferWrite(samples, count);
    }
}
```

### 环形缓冲区实现

```cpp
class RingBuffer {
private:
    int16_t* buffer;
    size_t size;
    size_t head;  // 写指针
    size_t tail;  // 读指针
    
public:
    RingBuffer(size_t capacity) {
        size = capacity;
        buffer = (int16_t*)ps_malloc(capacity * sizeof(int16_t));
        head = tail = 0;
    }
    
    void write(int16_t* data, size_t count) {
        for (size_t i = 0; i < count; i++) {
            buffer[head] = data[i];
            head = (head + 1) % size;
            
            // 缓冲区满时，推进读指针（覆盖旧数据）
            if (head == tail) {
                tail = (tail + 1) % size;
            }
        }
    }
    
    size_t read(int16_t* dest, size_t maxCount) {
        size_t count = 0;
        while (tail != head && count < maxCount) {
            dest[count++] = buffer[tail];
            tail = (tail + 1) % size;
        }
        return count;
    }
    
    size_t available() {
        return (head >= tail) ? (head - tail) : (size - tail + head);
    }
    
    void clear() {
        head = tail = 0;
    }
};

// 全局缓冲区（30 秒 @16kHz = 480,000 样本）
RingBuffer audioRingBuffer(480000);
```

### 设备端：构建 WAV 并上传

```cpp
// cloud_client.cpp

// WAV 头结构（44 字节）
struct WavHeader {
    char riff[4] = "RIFF";
    uint32_t riffSize;
    char wave[4] = "WAVE";
    char fmt[4] = "fmt ";
    uint32_t fmtSize = 16;
    uint16_t audioFormat = 1;     // PCM
    uint16_t numChannels = 1;     // 单声道
    uint32_t sampleRate = 16000;
    uint32_t byteRate = 32000;    // 16000 × 16/8 × 1
    uint16_t blockAlign = 2;      // 16/8 × 1
    uint16_t bitsPerSample = 16;
    char data[4] = "data";
    uint32_t dataSize;
};

// 上传 WAV 到后端
bool uploadToBackend(int16_t* pcmData, size_t sampleCount) {
    // 计算 PCM 大小
    uint32_t pcmSize = sampleCount * sizeof(int16_t);
    
    // 构建 WAV 头
    WavHeader header;
    header.riffSize = pcmSize + 36;  // 44 - 8
    header.dataSize = pcmSize;
    
    // 分配缓冲区（PSRAM）
    uint8_t* wavBuffer = (uint8_t*)ps_malloc(44 + pcmSize);
    memcpy(wavBuffer, &header, 44);
    memcpy(wavBuffer + 44, pcmData, pcmSize);
    
    // HTTP POST
    WiFiClientSecure client;
    client.setInsecure();  // 开发环境跳过证书验证
    
    if (!client.connect(BACKEND_HOST, BACKEND_PORT)) {
        Serial.println("[Cloud] 无法连接后端");
        free(wavBuffer);
        return false;
    }
    
    // 发送请求头
    client.print("POST /chat HTTP/1.1\r\n");
    client.print("Host: " + String(BACKEND_HOST) + "\r\n");
    client.print("Content-Type: audio/wav\r\n");
    client.print("Content-Length: " + String(44 + pcmSize) + "\r\n");
    client.print("X-Session-ID: " + deviceId + "\r\n");
    client.print("\r\n");
    
    // 发送 WAV 数据
    client.write(wavBuffer, 44 + pcmSize);
    
    // 等待响应
    uint32_t timeout = millis() + 30000;  // 30 秒超时
    while (!client.available() && millis() < timeout) {
        delay(10);
    }
    
    if (!client.available()) {
        Serial.println("[Cloud] 响应超时");
        free(wavBuffer);
        return false;
    }
    
    // 读取响应行
    String statusLine = client.readStringUntil('\n');
    Serial.println("[Cloud] 响应：" + statusLine);
    
    // 跳过响应头，找到空行
    while (client.available()) {
        String line = client.readStringUntil('\n');
        if (line == "\r" || line.length() == 0) break;
    }
    
    // 读取响应体（JSON）
    String responseBody = client.readString();
    Serial.println("[Cloud] 响应体：" + responseBody);
    
    // 解析 JSON
    parseBackendResponse(responseBody);
    
    free(wavBuffer);
    return true;
}
```

### 后端：处理 /chat 请求

```python
# main.py

from fastapi import FastAPI, Request, Header
from services import stt, llm, tts, emotion
from memory import repository
import uuid

app = FastAPI()

@app.post("/chat")
async def chat(
    request: Request,
    x_session_id: str | None = Header(None)
):
    """
    主聊天接口：接收音频，返回回复和 TTS 音频
    """
    # 1. 读取音频数据
    wav_bytes = await request.body()
    print(f"[Backend] 收到音频：{len(wav_bytes)} bytes")
    
    # 2. STT：语音转文字
    transcript = await stt.transcribe(wav_bytes)
    print(f"[Backend] STT 结果：{transcript}")
    
    # 3. 获取会话 ID
    session_id = x_session_id or "default"
    
    # 4. 获取记忆
    memory_summary = repository.get_memory_summary(session_id)
    session_history = repository.get_session_history(session_id)
    
    # 5. LLM：生成回复
    reply = await llm.chat_completion(
        transcript,
        memory_summary,
        session_history
    )
    print(f"[Backend] LLM 回复：{reply}")
    
    # 6. 情绪分析
    emo = emotion.analyze(reply)
    print(f"[Backend] 情绪：{emo}")
    
    # 7. TTS：生成音频
    audio_filename = f"{uuid.uuid4().hex}.wav"
    audio_url = await tts.generate_speech(reply, audio_filename)
    print(f"[Backend] TTS 音频：{audio_url}")
    
    # 8. 保存对话到记忆
    repository.save_interaction(session_id, transcript, reply)
    
    # 9. 返回响应
    return {
        "text_reply": reply,
        "emotion": emo,
        "audio_url": audio_url
    }
```

```python
        result = response.json()
        return result["text"]
```

### 后端：LLM 服务实现

```python
# services/llm.py

import httpx
from config import DASHSCOPE_KEY, DASHSCOPE_API_BASE, LLM_MODEL

SYSTEM_PROMPT = """你是一只可爱的电子猫咪，名叫"赛博喵"。
用简短的一两句话回复，可以加上喵~等拟声词。
你的性格：好奇、友好、偶尔撒娇。
"""

async def chat_completion(
    user_input: str,
    memory_summary: str = None,
    session_history: str = None
) -> str:
    """
    调用 LLM API 生成回复
    """
    # 构建消息
    messages = [{"role": "system", "content": SYSTEM_PROMPT}]
    
    # 注入记忆
    if memory_summary:
        messages[0]["content"] += f"\n[关于用户的记忆：{memory_summary}]"
    
    # 注入对话历史（最近 10 轮）
    if session_history:
        messages.append({
            "role": "user",
            "content": f"[最近对话]\n{session_history}"
        })
    
    # 用户输入
    messages.append({"role": "user", "content": user_input})
    
    async with httpx.AsyncClient() as client:
        response = await client.post(
            DASHSCOPE_API_BASE,
            headers={"Authorization": f"Bearer {DASHSCOPE_KEY}"},
            json={
                "model": LLM_MODEL,
                "messages": messages
            },
            timeout=30
        )
        
        if response.status_code != 200:
            raise Exception(f"LLM API 错误：{response.status_code}")
        
        result = response.json()
        return result["choices"][0]["message"]["content"]
```

```cpp
    lv_anim_set_exec_cb(&a, _setEyeRotation);
```

```cpp
// Talking 表情有 4 段循环律动

void CatFace::_startTalkingAnim() {
    _talkStage = 0;
    _talkEpoch++;  // 纪元计数器，防止残留回调
    _scheduleTalkStage();
}

void CatFace::_scheduleTalkStage() {
    lv_anim_t a;
    lv_anim_init(&a);
    lv_anim_set_var(&a, _leftEye);
    
    int32_t targetW, targetH, targetRot;
    uint32_t duration;
    
    switch (_talkStage) {
        case 0:  // 挤压上提
            targetW = 100; targetH = 60; targetRot = -100;
            duration = 100;
            break;
        case 1:  // 展开下沉
            targetW = 95; targetH = 95; targetRot = -50;
            duration = 150;
            break;
        case 2:  // 轻微过冲
            targetW = 98; targetH = 98; targetRot = 0;
            duration = 100;
            break;
        case 3:  // 归位
            targetW = 95; targetH = 95; targetRot = 0;
            duration = 100;
            break;
    }
    
    lv_anim_set_values(&a, _currentEye.w, targetW);
    lv_anim_set_duration(&a, duration);
    lv_anim_set_path_cb(&a, lv_anim_path_ease_out);
    lv_anim_set_exec_cb(&a, _setEyeWidth);
    lv_anim_set_completed_cb(&a, _talkStageDone);
    lv_anim_start(&a);
    
    // 高度和旋转类似...
}

void CatFace::_talkStageDone(lv_anim_t* a) {
    _talkStage = (_talkStage + 1) % 4;
    _scheduleTalkStage();
}
```

## 常见问题

**Q: 录音数据全是 0 怎么办？**  
A: 检查 ES7210 初始化序列，特别是 REG01 时钟配置。确认 I2S 引脚配置正确。

**Q: 后端响应超时？**  
A: 检查服务器 IP/端口，确认后端服务在运行。检查防火墙规则。

**Q: 表情切换卡顿？**  
A: 确保 LVGL 任务优先级足够高，检查 `lv_timer_handler()` 是否每 16ms 调用。

**Q: TTS 播放速度异常？**  
A: 确认 TTS 输出的采样率（16kHz）与 I2S 配置一致。检查 mono→stereo 转换。
