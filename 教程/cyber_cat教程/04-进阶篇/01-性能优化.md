## 4.1.1 固件端性能优化

### 低功耗设计

ESP32-S3 支持多种功耗模式，合理使用可以显著延长电池寿命。

#### ESP32-S3 功耗模式

| 模式 | 电流 | 唤醒源 | 适用场景 |
|------|------|--------|----------|
| Active | ~150mA | - | 正常运行（屏幕亮、WiFi 连接） |
| Modem Sleep | ~50mA | - | 短时待机（关闭射频） |
| Light Sleep | ~15mA | GPIO、触摸、RTC 定时器 | 中等待机（CPU 暂停） |
| Deep Sleep | ~0.1mA | GPIO、RTC 定时器、ULP 协处理器 | 长时待机（RAM 断电） |

```cpp
    }
```

```cpp
        Serial.println("[DeepSleep] 准备进入深度睡眠...");
        
        // 保存关键数据到 RTC 内存
        saveStateToRTC();
        
        // 配置唤醒源
        configureWakeUpSources();
        
        // 配置定时器
        esp_sleep_enable_timer_wakeup(sleepDurationSeconds * uS_TO_S_FACTOR);
        
        // 关闭外设
```

```cpp
// src/memory_manager.cpp
#include <esp_heap_caps.h>

class MemoryManager {
public:
    // 内存分配统计
    struct MemoryStats {
        size_t totalPsram;
        size_t freePsram;
        size_t largestFreeBlock;
        size_t totalDram;
        size_t freeDram;
    };
    
    static MemoryStats getStats() {
        MemoryStats stats;
        
        stats.totalPsram = heap_caps_get_total_size(MALLOC_CAP_SPIRAM);
        stats.freePsram = heap_caps_get_free_size(MALLOC_CAP_SPIRAM);
        stats.largestFreeBlock = heap_caps_get_largest_free_block(MALLOC_CAP_SPIRAM);
        
        stats.totalDram = heap_caps_get_total_size(MALLOC_CAP_INTERNAL);
        stats.freeDram = heap_caps_get_free_size(MALLOC_CAP_INTERNAL);
        
        return stats;
    }
    
    static void printStats() {
        MemoryStats stats = getStats();
        
        Serial.println("=== 内存统计 ===");
        Serial.printf("PSRAM: 总计 %d KB, 空闲 %d KB, 最大连续块 %d KB\n",
                      stats.totalPsram / 1024,
                      stats.freePsram / 1024,
                      stats.largestFreeBlock / 1024);
        Serial.printf("DRAM:  总计 %d KB, 空闲 %d KB\n",
                      stats.totalDram / 1024,
                      stats.freeDram / 1024);
    }
    
    // PSRAM 分配包装器
    template<typename T>
    static T* psAlloc(size_t count = 1) {
        size_t size = sizeof(T) * count;
        void* ptr = heap_caps_malloc(size, MALLOC_CAP_SPIRAM);
        if (!ptr) {
            Serial.printf("[Memory] PSRAM 分配失败：%d bytes\n", size);
            return nullptr;
        }
        memset(ptr, 0, size);  // 初始化为 0
        return reinterpret_cast<T*>(ptr);
    }
    
    static void psFree(void* ptr) {
        if (ptr) {
            heap_caps_free(ptr);
        }
    }
};

// 使用示例
```

#### 内存碎片整理

```cpp
// memory_manager.cpp 增强版

class MemoryManager {
public:
    // 检测内存碎片
    static float getFragmentationRatio() {
        size_t total = heap_caps_get_total_size(MALLOC_CAP_SPIRAM);
        size_t free = heap_caps_get_free_size(MALLOC_CAP_SPIRAM);
        size_t largest = heap_caps_get_largest_free_block(MALLOC_CAP_SPIRAM);
        
        if (free == 0) return 0.0f;
        
        // 碎片率 = (总空闲 - 最大连续块) / 总空闲
        float fragmentation = (float)(free - largest) / free;
        return fragmentation;
    }
    
    // 内存分配策略建议
    static const char* getAllocationAdvice() {
        float frag = getFragmentationRatio();
        
        if (frag < 0.1f) {
            return "内存状态良好";
        } else if (frag < 0.3f) {
            return "轻度碎片，注意大对象分配";
        } else if (frag < 0.5f) {
            return "中度碎片，考虑重启或整理";
        } else {
            return "严重碎片，建议重启设备";
        }
    }
    
    // 定期整理（软重启）
    static void scheduleSoftRestart() {
        Serial.println("[Memory] 计划软重启以整理内存碎片");
        
        // 保存状态到 RTC 内存
        RTC_DATA_ATTR bool softRestartPending = true;
        
        // 5 秒后重启
        esp_sleep_enable_timer_wakeup(5 * uS_TO_S_FACTOR);
        esp_sleep_pd_config(ESP_PD_DOMAIN_RTC_PERIPH, ESP_PD_OPTION_OFF);
        esp_deep_sleep_start();
    }
};

// 主循环中检查
void loop() {
    // 每 60 秒检查一次内存
    static uint32_t lastCheck = 0;
    if (millis() - lastCheck > 60000) {
        float frag = MemoryManager::getFragmentationRatio();
        Serial.printf("[Memory] 碎片率：%.1f%%\n", frag * 100);
        
        if (frag > 0.5f) {
            MemoryManager::scheduleSoftRestart();
        }
        
        lastCheck = millis();
    }
    
    // ... 正常业务逻辑
}
```

```cpp
// src/object_pool.h
#pragma once
#include <Arduino.h>

// 通用对象池模板
template<typename T, size_t SIZE>
class ObjectPool {
private:
    T _objects[SIZE];
    bool _inUse[SIZE];
    
public:
    ObjectPool() {
        memset(_inUse, 0, SIZE);
    }
    
    T* acquire() {
        for (size_t i = 0; i < SIZE; i++) {
            if (!_inUse[i]) {
                _inUse[i] = true;
                return &_objects[i];
            }
        }
        Serial.printf("[Pool] 对象池已满 (max=%d)\n", SIZE);
        return nullptr;
    }
    
    void release(T* obj) {
        for (size_t i = 0; i < SIZE; i++) {
            if (&_objects[i] == obj) {
                _inUse[i] = false;
                return;
            }
        }
    }
    
    size_t getInUseCount() {
        size_t count = 0;
        for (size_t i = 0; i < SIZE; i++) {
            if (_inUse[i]) count++;
        }
        return count;
    }
};

// 使用示例：音频缓冲区池
struct AudioBuffer {
    int16_t samples[8192];
    size_t sampleCount;
```

```python
# memory/db_pool.py
import sqlite3
from queue import Queue
from contextlib import contextmanager
import threading

class ConnectionPool:
    def __init__(self, db_path: str, max_connections: int = 10):
        self.db_path = db_path
        self.max_connections = max_connections
        self._pool = Queue(maxsize=max_connections)
        self._lock = threading.Lock()
        self._created = 0
        
        # 预创建连接
        for _ in range(min(4, max_connections)):
            conn = sqlite3.connect(db_path, check_same_thread=False)
            conn.row_factory = sqlite3.Row
            self._pool.put(conn)
    
    @contextmanager
    def get_connection(self):
        """获取连接的上下文管理器"""
        conn = None
        try:
            # 尝试从池中获取
            conn = self._pool.get(timeout=5.0)
            yield conn
        except Queue.Empty:
            # 池为空，创建临时连接
            with self._lock:
                if self._created < self.max_connections:
                    conn = sqlite3.connect(self.db_path, check_same_thread=False)
                    conn.row_factory = sqlite3.Row
                    self._created += 1
                    yield conn
                else:
                    raise RuntimeError("数据库连接池已满")
        finally:
            if conn:
                self._pool.put(conn)
    
    def close_all(self):
        """关闭所有连接"""
        while not self._pool.empty():
            conn = self._pool.get()
            conn.close()
```

```python
# services/async_http.py
import aiohttp
import asyncio

class AsyncHttpClient:
    def __init__(self, max_connections: int = 10):
        self._session = None
        self._max_connections = max_connections
        self._semaphore = asyncio.Semaphore(max_connections)
    
    async def get_session(self) -> aiohttp.ClientSession:
        if self._session is None or self._session.closed:
            timeout = aiohttp.ClientTimeout(total=30)
            connector = aiohttp.TCPConnector(
                limit=self._max_connections,
                ttl_dns_cache=300,
                use_dns_cache=True,
            )
            self._session = aiohttp.ClientSession(
                timeout=timeout,
                connector=connector
            )
        return self._session
    
    async def post_json(self, url: str, data: dict, headers: dict = None) -> dict:
        async with self._semaphore:
            session = await self.get_session()
            async with session.post(url, json=data, headers=headers) as resp:
                resp.raise_for_status()
                return await resp.json()
    
    async def post_bytes(self, url: str, data: bytes, headers: dict = None) -> bytes:
        async with self._semaphore:
            session = await self.get_session()
            async with session.post(url, data=data, headers=headers) as resp:
                resp.raise_for_status()
                return await resp.read()
    
    async def close(self):
        if self._session and not self._session.closed:
            await self._session.close()
```

```python
        self._api_key = SILICONFLOW_KEY
        self._base_url = "https://api.siliconflow.cn/v1"
    
    async def chat_completion_stream(
        self,
        user_input: str,
        memory_summary: str = None
    ) -> AsyncGenerator[str, None]:
        """流式 LLM 响应"""
        
        messages = self._build_messages(user_input, memory_summary)
        
        payload = {
            "model": "alibaba/Qwen2.5-Max-1M",
            "messages": messages,
            "max_tokens": 300,
            "temperature": 0.7,
            "stream": True  # 启用流式
        }
        
        headers = {
            "Authorization": f"Bearer {self._api_key}",
            "Content-Type": "application/json"
        }
        
        async with aiohttp.ClientSession() as session:
            async with session.post(
                f"{self._base_url}/chat/completions",
                json=payload,
                headers=headers
            ) as resp:
                async for line in resp.content:
                    line = line.decode('utf-8').strip()
                    if line.startswith('data: '):
                        data = line[6:]
                        if data == '[DONE]':
                            break
                        try:
                            chunk = json.loads(data)
                            delta = chunk['choices'][0]['delta']
                            if 'content' in delta:
                                yield delta['content']
                        except json.JSONDecodeError:
                            continue
```

```cpp
class StreamingAudioPlayer {
private:
    static const size_t BUFFER_SIZE = 4096;
    uint8_t* _buffer;
    size_t _bufferPos;
    bool _isPlaying;
    
    i2s_config_t _i2sConfig;
    
public:
    void init() {
        _buffer = (uint8_t*)ps_malloc(BUFFER_SIZE);
        _bufferPos = 0;
        _isPlaying = false;
        
        // 配置 I2S
        _i2sConfig = {
            .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_TX),
            .sample_rate = 16000,
            .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
            .channel_format = I2S_CHANNEL_FMT_ONLY_RIGHT,
            .communication_format = I2S_COMM_FORMAT_STAND_I2S,
            .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
            .dma_buf_count = 4,
            .dma_buf_len = 256,
        };
        
        i2s_driver_install(I2S_NUM_0, &_i2sConfig, 0, NULL);
    }
    
    bool playFromUrl(const char* url) {
        Serial.printf("[Player] 播放 URL: %s\n", url);
        
        WiFiClientSecure client;
        if (!client.connect(URL_HOST, 443)) {
            Serial.println("[Player] 连接失败");
            return false;
        }
        
        // HTTP GET 请求
        client.print(String("GET ") + url + " HTTP/1.1\r\n");
        client.print("Host: " + String(URL_HOST) + "\r\n");
        client.print("Connection: close\r\n\r\n");
        
        // 跳过响应头
        while (client.connected()) {
            String line = client.readLineUntil('\n');
            if (line == "\r") break;
        }
        
        // 跳过 WAV 头（44 字节）
        client.read(_buffer, 44);
        
        // 流式播放：边下载边播放
        _isPlaying = true;
        size_t totalBytes = 0;
        
        while (client.connected() && _isPlaying) {
            int available = client.available();
            if (available > 0) {
                // 读取一批数据
                size_t toRead = min(available, BUFFER_SIZE);
                int bytesRead = client.read(_buffer, toRead);
                
                // 发送到 I2S
                size_t bytesWritten = 0;
                i2s_write(I2S_NUM_0, _buffer, bytesRead, &bytesWritten, portMAX_DELAY);
                
                totalBytes += bytesRead;
                
                if (totalBytes % 16384 == 0) {
                    Serial.printf("[Player] 已播放 %d KB\n", totalBytes / 1024);
                }
            }
            delay(1);
        }
        
        Serial.printf("[Player] 播放完成，共 %d KB\n", totalBytes / 1024);
        return true;
    }
    
    void stop() {
        _isPlaying = false;
    }
};
```

```python
# middleware/performance.py
import time
import psutil
from fastapi import Request, Response
from starlette.middleware.base import BaseHTTPMiddleware

class PerformanceMiddleware(BaseHTTPMiddleware):
    def __init__(self, app, log_threshold_ms: int = 1000):
        super().__init__(app)
        self.log_threshold_ms = log_threshold_ms
        self.request_count = 0
        self.total_time_ms = 0
    
    async def dispatch(self, request: Request, call_next) -> Response:
        start_time = time.time()
        
        # 处理请求
        response = await call_next(request)
        
        # 计算耗时
        duration_ms = (time.time() - start_time) * 1000
        self.request_count += 1
        self.total_time_ms += duration_ms
        
        # 记录慢请求
        if duration_ms > self.log_threshold_ms:
            print(f"[Perf] 慢请求：{request.url.path} - {duration_ms:.1f}ms")
        
        # 添加响应头
        response.headers["X-Response-Time"] = f"{duration_ms:.1f}ms"
        
        return response
    
    def get_stats(self) -> dict:
        return {
            "request_count": self.request_count,
            "avg_response_time_ms": self.total_time_ms / max(self.request_count, 1),
            "memory_percent": psutil.Process().memory_percent(),
            "cpu_percent": psutil.Process().cpu_percent(),
        }

# 使用示例
# main.py
app.add_middleware(PerformanceMiddleware, log_threshold_ms=1000)

@app.get("/health")
async def health():
    stats = app.user_middleware[0].get_stats()
    return {
        **stats,
        "status": "healthy"
    }
```
