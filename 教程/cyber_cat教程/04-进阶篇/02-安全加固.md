## 4.2.1 API 鉴权

### 设备端 Token 认证

固件端在请求后端 API 时，需要携带认证 Token 防止未授权访问。

```cpp

class AuthClient {

    // 刷新 Token
    bool refreshToken();
    
    // 添加认证头到 HTTP 请求
    void addAuthHeader(WiFiClientSecure& client);
```

```cpp
// src/auth_client.cpp
#include "auth_client.h"
#include <HTTPClient.h>
#include <ArduinoJson.h>

AuthClient::AuthClient(const String& deviceId, const String& initialToken)
    : _deviceId(deviceId),
      _deviceToken(initialToken),
      _tokenExpiresAt(0),
      _tokenValid(initialToken.length() > 0) {
}

void AuthClient::begin(const String& backendHost, uint16_t backendPort) {
    _backendHost = backendHost;
    _backendPort = backendPort;
    
    Serial.printf("[Auth] 设备 ID: %s\n", _deviceId.c_str());
    
    // 初次启动，刷新 Token
    if (!_tokenValid) {
        refreshToken();
    }
}

const String& AuthClient::getToken() {
    // Token 即将过期（剩余 5 分钟）时提前刷新
    if (_tokenValid && millis() / 1000 > _tokenExpiresAt - 300) {
        Serial.println("[Auth] Token 即将过期，刷新中...");
        refreshToken();
    }
    return _deviceToken;
}

bool AuthClient::refreshToken() {
    Serial.println("[Auth] 刷新 Token...");
    
    WiFiClientSecure client;
    if (!client.connect(_backendHost.c_str(), _backendPort)) {
        Serial.println("[Auth] 无法连接后端");
        return false;
    }
    
    // 构建 Token 刷新请求
    String url = "/auth/refresh";
    
    client.print(String("POST ") + url + " HTTP/1.1\r\n");
    client.print("Host: " + _backendHost + "\r\n");
    client.print("Content-Type: application/json\r\n");
    client.print("Content-Length: 0\r\n");
    
    // 使用旧 Token（如果有）
    if (_tokenValid && _deviceToken.length() > 0) {
        client.print("Authorization: Bearer " + _deviceToken + "\r\n");
    }
    
    // 设备 ID 用于识别
    client.print("X-Device-ID: " + _deviceId + "\r\n");
    
    client.print("\r\n");
    
    // 读取响应
    delay(100);  // 等待响应
    
    String response = "";
    while (client.connected()) {
        String line = client.readStringUntil('\n');
        if (line == "\r") {
            // 响应头结束，读取正文
            response = client.readString();
            break;
        }
    }
    
    client.stop();
    
    // 解析 JSON 响应
    StaticJsonDocument<256> doc;
    DeserializationError err = deserializeJson(doc, response);
    
    if (err) {
        Serial.printf("[Auth] JSON 解析失败：%s\n", err.c_str());
        return false;
    }
    
    const char* newToken = doc["token"];
    int expiresIn = doc["expires_in"];  // 秒
    
    if (newToken && expiresIn > 0) {
        _deviceToken = String(newToken);
        _tokenExpiresAt = millis() / 1000 + expiresIn;
        _tokenValid = true;
        
        Serial.printf("[Auth] Token 刷新成功，有效期 %d 秒\n", expiresIn);
        return true;
    }
    
    Serial.println("[Auth] Token 刷新失败");
    return false;
}

void AuthClient::addAuthHeader(WiFiClientSecure& client) {
    String token = getToken();
    if (_tokenValid && token.length() > 0) {
        client.print("Authorization: Bearer " + token + "\r\n");
    }
    client.print("X-Device-ID: " + _deviceId + "\r\n");
}
```

### 后端 Token 验证

```python
# middleware/auth.py
import jwt
import time
from functools import wraps
from fastapi import Request, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials

# JWT 配置
JWT_SECRET = "your-secret-key-change-in-production"  # 生产环境用环境变量
JWT_ALGORITHM = "HS256"
TOKEN_EXPIRY_SECONDS = 3600  # 1 小时

security = HTTPBearer(auto_error=False)

def create_token(device_id: str, expires_in: int = TOKEN_EXPIRY_SECONDS) -> str:
    """生成 JWT Token"""
    expire = int(time.time()) + expires_in
    payload = {
        "device_id": device_id,
        "exp": expire,
        "iat": int(time.time()),
        "type": "device"
    }
    token = jwt.encode(payload, JWT_SECRET, algorithm=JWT_ALGORITHM)
    return token

def verify_token(token: str) -> dict:
    """验证 JWT Token"""
    try:
        payload = jwt.decode(token, JWT_SECRET, algorithms=[JWT_ALGORITHM])
        return {
            "valid": True,
            "device_id": payload.get("device_id"),
            "exp": payload.get("exp")
        }
    except jwt.ExpiredSignatureError:
        return {"valid": False, "error": "token_expired"}
    except jwt.InvalidTokenError:
        return {"valid": False, "error": "invalid_token"}

def require_auth(func):
    """认证装饰器"""
    @wraps(func)
    async def wrapper(request: Request, *args, **kwargs):
        # 获取 Authorization 头
        credentials = await security(request)
        
        if not credentials:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Missing authentication token"
            )
        
        # 验证 Token
        result = verify_token(credentials.credentials)
        
        if not result["valid"]:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail=f"Invalid token: {result.get('error')}"
            )
        
        # 添加设备 ID 到请求上下文
        request.state.device_id = result["device_id"]
        
        return await func(request, *args, **kwargs)
    
    return wrapper

# 使用示例
# main.py
from middleware.auth import require_auth, create_token

    """刷新 Token 接口"""

        raise HTTPException(status_code=400, detail="Missing X-Device-ID")
    
    # 生成新 Token
    token = create_token(device_id)
    
    return {
        "token": token,
        "expires_in": TOKEN_EXPIRY_SECONDS
    }

@app.post("/chat")
```

### 设备注册与激活

```python
# services/device_auth.py
import sqlite3
import secrets
from datetime import datetime, timedelta

class DeviceAuthService:
    def __init__(self):
        self.db_path = "/var/lib/cybercat/auth.db"
        self._init_db()
    
    def _init_db(self):
        """初始化认证数据库"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS devices (
                device_id TEXT PRIMARY KEY,
                device_name TEXT,
                device_secret TEXT,
                status TEXT DEFAULT 'pending',  -- pending, active, revoked
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                last_seen_at TIMESTAMP,
                activated_at TIMESTAMP
            )
        """)
        
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS token_blacklist (
                token_hash TEXT PRIMARY KEY,
                blacklisted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                reason TEXT
            )
        """)
        
        conn.commit()
        conn.close()
    
    def register_device(self, device_id: str, device_name: str = None) -> dict:
        """注册新设备（需要管理员审核）"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # 生成设备密钥（用于初始激活）
        device_secret = secrets.token_urlsafe(32)
        
        try:
            cursor.execute("""
                INSERT OR REPLACE INTO devices 
                (device_id, device_name, device_secret, status)
                VALUES (?, ?, ?, 'pending')
            """, (device_id, device_name, device_secret))
            
            conn.commit()
            
            return {
                "device_id": device_id,
                "device_secret": device_secret,
                "status": "pending",
                "message": "设备已注册，等待管理员激活"
            }
        finally:
            conn.close()
    
    def activate_device(self, device_id: str, device_secret: str) -> bool:
        """管理员激活设备"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            UPDATE devices 
            SET status = 'active', activated_at = CURRENT_TIMESTAMP
            WHERE device_id = ? AND device_secret = ? AND status = 'pending'
        """, (device_id, device_secret))
        
        affected = cursor.rowcount
        conn.commit()
        conn.close()
        
        return affected > 0
    
    def revoke_device(self, device_id: str, reason: str = None) -> bool:
        """吊销设备"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            UPDATE devices SET status = 'revoked'
            WHERE device_id = ?
        """, (device_id,))
        
        # 将当前 Token 加入黑名单
        # (实际使用时需要配合 Redis 等缓存实现)
        
        conn.commit()
        conn.close()
        
        return True
    
    def is_device_authorized(self, device_id: str) -> bool:
        """检查设备是否已授权"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            SELECT status FROM devices WHERE device_id = ?
        """, (device_id,))
        
        row = cursor.fetchone()
        conn.close()
        
        return row and row[0] == 'active'
    
    def update_last_seen(self, device_id: str):
        """更新设备最后在线时间"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            UPDATE devices SET last_seen_at = CURRENT_TIMESTAMP
            WHERE device_id = ?
        """, (device_id,))
        
        conn.commit()
        conn.close()
```

## 4.2.2 数据加密

### 敏感数据加密存储

```python
# utils/encryption.py
from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
import base64
import os

class DataEncryption:
    def __init__(self, master_key: str = None):
        """
        初始化加密器
        
        Args:
            master_key: 主密钥，如果为 None 则生成随机密钥
        """
        self.master_key = master_key or os.environ.get("ENCRYPTION_KEY")
        
        if not self.master_key:
            # 生成新密钥（生产环境应该保存密钥到安全位置）
            self.master_key = Fernet.generate_key().decode()
            print(f"[Encryption] 生成新密钥：{self.master_key}")
            print("[Encryption] 警告：请保存密钥到 ENCRYPTION_KEY 环境变量")
    
    def _derive_key(self, salt: bytes) -> bytes:
        """从主密钥派生加密密钥"""
        kdf = PBKDF2HMAC(
            algorithm=hashes.SHA256(),
            length=32,
            salt=salt,
            iterations=100000,
        )
        return base64.urlsafe_b64encode(kdf.derive(self.master_key.encode()))
    
    def encrypt(self, plaintext: str) -> str:
        """加密字符串"""
        salt = os.urandom(16)
        key = self._derive_key(salt)
        f = Fernet(key)
        
        encrypted = f.encrypt(plaintext.encode())
        
        # 编码为 base64 字符串（salt + encrypted）
        result = base64.b64encode(salt + encrypted).decode()
        return result
    
    def decrypt(self, ciphertext: str) -> str:
        """解密字符串"""
        try:
            # 解码 base64
            data = base64.b64decode(ciphertext.encode())
            
            # 提取 salt 和密文
            salt = data[:16]
            encrypted = data[16:]
            
            # 派生密钥并解密
            key = self._derive_key(salt)
            f = Fernet(key)
            
            decrypted = f.decrypt(encrypted)
            return decrypted.decode()
        except Exception as e:
            print(f"[Encryption] 解密失败：{e}")
            return None

# 使用示例
# memory/repository.py
from utils.encryption import DataEncryption

encryptor = DataEncryption()

def save_interaction(session_id: str, transcript: str, response: str):
    """保存对话（加密敏感数据）"""
    conn = get_connection()
    cursor = conn.cursor()
    
    # 加密用户输入（可能包含个人信息）
    encrypted_transcript = encryptor.encrypt(transcript)
    
    # 回复不加密（用于情绪分析等）
    cursor.execute("""
        INSERT INTO interactions (session_id, transcript, response, created_at)
        VALUES (?, ?, ?, CURRENT_TIMESTAMP)
    """, (session_id, encrypted_transcript, response))
    
    conn.commit()
    conn.close()

def get_transcript(interaction_id: int) -> str:
    """获取解密后的用户输入"""
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute("SELECT transcript FROM interactions WHERE id = ?", (interaction_id,))
    row = cursor.fetchone()
    conn.close()
    
    if row and row[0]:
        return encryptor.decrypt(row[0])
    return None
```

```python
# middleware/input_validation.py
from pydantic import BaseModel, Field, validator
import re
from fastapi import HTTPException

class ChatRequest(BaseModel):
    """聊天请求模型（带验证）"""
    transcript: str = Field(..., min_length=1, max_length=1000)
    device_id: str = Field(..., pattern=r"^[a-zA-Z0-9-]+$")
    
    @validator('transcript')
    def validate_transcript(cls, v):
        # 过滤潜在注入攻击
        dangerous_patterns = [
            r"<script",      # XSS
            r"javascript:",  # 协议注入
            r"union\s+select",  # SQL 注入
            r";\s*drop\s+table",  # SQL 注入
            r"\.\./\.\.",    # 路径遍历
        ]
        
        for pattern in dangerous_patterns:
            if re.search(pattern, v, re.IGNORECASE):
                raise ValueError(f"输入包含潜在危险内容")
        
        # 限制中文字符数量（防止长文本攻击）
        chinese_chars = len(re.findall(r'[\u4e00-\u9fff]', v))
        if chinese_chars > 500:
            raise ValueError("中文内容过长")
        
        return v
```
