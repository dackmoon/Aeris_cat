## 3.3.1 任务描述

### 目标

支持 **多个 ESP32 设备同时连接后端**。

### 场景

- 家里有多个猫咪设备
- 每个设备有独立的记忆
- 后端能区分不同设备

### 技术方案

- 使用 ESP32 的 MAC 地址作为设备 ID
- 后端按设备 ID 隔离记忆
- 支持设备命名（"小喵 1 号"、"小喵 2 号"）

### 前置知识

- WiFi 基础
- HTTP 请求头
- 会话管理

## 3.3.2 步骤 1：ESP32 端添加设备 ID

### 获取 MAC 地址

每个 ESP32 都有唯一的 MAC 地址，可作为设备标识：

```cpp
// src/cloud_client.cpp

#include <WiFi.h>

void CloudClient::begin(const char* apiKey, ...) {
    // 获取 MAC 地址作为设备 ID
    String macAddress = WiFi.macAddress();
    Serial.printf("[Cloud] 设备 ID: %s\n", macAddress.c_str());

    // 在 HTTP 请求头中传递设备 ID
    http.addHeader("X-Device-ID", macAddress);

    // 也可以用 session_id 参数
    _sessionId = "device-" + macAddress;
}
```

### 完整实现

```cpp
// cloud_client.cpp

#include "cloud_client.h"

CloudClient::CloudClient() {
}

void CloudClient::begin() {
    // 获取 MAC 地址作为设备 ID
    // 格式：XX:XX:XX:XX:XX:XX
    _deviceId = WiFi.macAddress();
    
    // 移除冒号，变成纯字母数字
    _deviceId.replace(":", "");
    _deviceId.toLowerCase();
    
    Serial.printf("[Cloud] 设备 ID: %s\n", _deviceId.c_str());
    
    // 生成初始会话 Token
    generateSessionToken();
}

void CloudClient::generateSessionToken() {
    // 简单 Token：设备 ID + 时间戳
    _sessionToken = _deviceId + "_" + String(millis());
}

bool CloudClient::uploadAudio(int16_t* pcmData, size_t sampleCount) {
    if (!_client.connect(BACKEND_HOST, BACKEND_PORT)) {
        Serial.println("[Cloud] 无法连接后端");
        return false;
    }
    
    // 构建 WAV 数据
    uint32_t pcmSize = sampleCount * sizeof(int16_t);
    uint32_t wavSize = 44 + pcmSize;
    
    uint8_t* wavBuffer = (uint8_t*)ps_malloc(wavSize);
    if (!wavBuffer) {
        Serial.println("[Cloud] PSRAM 不足");
        return false;
    }
    
    // 填充 WAV 头（略）
    // ...
    memcpy(wavBuffer + 44, pcmData, pcmSize);
    
    // 发送 HTTP 请求
    _client.print("POST /chat HTTP/1.1\r\n");
    _client.print("Host: " + String(BACKEND_HOST) + "\r\n");
    _client.print("Content-Type: audio/wav\r\n");
    _client.print("Content-Length: " + String(wavSize) + "\r\n");
    
    // 关键：添加设备 ID 请求头
    _client.print("X-Device-ID: " + _deviceId + "\r\n");
    _client.print("X-Session-Token: " + _sessionToken + "\r\n");
    
    _client.print("\r\n");
    _client.write(wavBuffer, wavSize);
    
    free(wavBuffer);
    
    // 读取响应（略）
    // ...
    
    return true;
}
```

## 3.3.3 步骤 2：后端按设备隔离记忆

### 后端：从请求头获取设备 ID

```python
# main.py

from fastapi import FastAPI, Request, Header
from typing import Optional

app = FastAPI()

@app.post("/chat")
async def chat(
    request: Request,
    x_device_id: Optional[str] = Header(None),
    x_session_token: Optional[str] = Header(None)
):
    """
    主聊天接口
    
    请求头：
    - X-Device-ID: 设备标识（可选）
    - X-Session-Token: 会话 Token（可选）
    """
    # 1. 读取音频数据
    wav_bytes = await request.body()
    
    # 2. 确定会话 ID
    if x_device_id:
        # 有设备 ID，使用该设备的独立会话
        session_id = f"device-{x_device_id}"
    else:
        # 无设备 ID，使用默认会话
        session_id = "default"
    
    Serial.println(f"[Backend] 会话 ID: {session_id}")
    
    # 3. STT
    transcript = await stt.transcribe(wav_bytes)
    
    # 4. 获取该设备的记忆
    memory_summary = repository.get_memory_summary(session_id)
    session_history = repository.get_session_history(session_id)
    
    # 5. LLM（注入记忆）
    reply = await llm.chat_completion(
        transcript,
        memory_summary=memory_summary,
        session_history=session_history
    )
    
    # 6. 情绪分析
    emo = emotion.analyze(reply)
    
    # 7. TTS
    import uuid
    audio_filename = f"{uuid.uuid4().hex}.wav"
    audio_url = await tts.generate_speech(reply, audio_filename)
    
    # 8. 保存对话到该设备的记忆
    repository.save_interaction(session_id, transcript, reply)
    
    return {
        "text_reply": reply,
        "emotion": emo,
        "audio_url": audio_url,
        "device_id": session_id  # 返回设备 ID 用于调试
    }
```

### 设备注册（可选）

```python
# memory/repository.py

def register_device(device_id: str, name: str = None, location: str = None):
    """注册设备信息"""
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute("""
        INSERT OR REPLACE INTO devices (device_id, name, location, last_seen)
        VALUES (?, ?, ?, CURRENT_TIMESTAMP)
    """, (device_id, name, location))
    
    conn.commit()
    conn.close()

def get_device_info(device_id: str) -> dict:
    """获取设备信息"""
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute("""
        SELECT device_id, name, location, last_seen
        FROM devices
        WHERE device_id = ?
    """, (device_id,))
    
    row = cursor.fetchone()
    conn.close()
    
    return dict(row) if row else None
```

### 后端记录设备信息

```python
# memory/repository.py

def get_all_devices() -> list:
    """获取所有设备列表"""
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute("""
        SELECT device_id, name, location, last_seen
        FROM devices
        ORDER BY last_seen DESC
    """)
    
    devices = [dict(row) for row in cursor.fetchall()]
    conn.close()
    
    return devices

def update_device_name(device_id: str, name: str):
    """更新设备名称"""
    conn = get_connection()
    cursor = conn.cursor()
    
    cursor.execute("""
        UPDATE devices
        SET name = ?
        WHERE device_id = ?
    """, (name, device_id))
    
    conn.commit()
    conn.close()
```

## 常见问题

**Q: 设备 ID 会变化吗？**  
A: WiFi MAC 地址是固定的，不会变化。但如果重新烧录 efuse，可能会改变。

**Q: 如何重置设备 ID？**  
A: 可以在代码中自定义 ID，或使用 `esp_efuse_mac_get_default()` 获取出厂 MAC。

**Q: 设备太多导致数据库过大？**  
A: 实现数据归档机制，将过期设备的记忆移入冷存储。
