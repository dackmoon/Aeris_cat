## 3.2.1 任务描述

### 目标

实现 **离线语音命令识别**（无需调用后端）。

### 支持的命令

| 命令词 | 触发动作 |
|--------|----------|
| "开灯" | 打开屏幕背光 |
| "关灯" | 关闭屏幕背光 |
| "几点了" | 显示当前时间 |
| "休息一下" | 进入睡眠模式 |

### 技术方案

- 使用 ESP32 的语音识别库（ESP-SR）
- 本地识别，无需联网
- 低功耗待机唤醒

### 前置知识

- ESP32 音频输入基础
- FreeRTOS 任务创建
- 状态机设计

## 3.2.2 步骤 1：安装 ESP-SR 库

### 编辑 `platformio.ini`

添加 ESP-SR 库依赖：

```ini
; platformio.ini

[env:esp32s3]

board = esp32-s3-devkitc-1
framework = arduino

lib_deps =
    lvgl/lvgl@~9.3.0
    bblanchon/ArduinoJson@^7
    esp-sr@^1.0.0              ; ← 新增 ESP 语音识别库

build_flags =
    -D ARDUINO_USB_CDC_ON_BOOT=1
    -D BOARD_HAS_PSRAM
    -D LV_CONF_INCLUDE_SIMPLE
    -I src
    -D CONFIG_ESP_SR_LIB_ENABLE  ; 启用 ESP-SR
```

### 安装依赖

```bash
cd /home/workspace/Aeris/cyber_cat
pio lib install esp-sr
```

### 下载模型文件

ESP-SR 需要语音识别模型文件：

```bash
# 创建模型目录
mkdir -p /sdcard/sr_model

# 从 ESP-SR 仓库下载模型
# 或从本项目 releases 页面下载
wget https://github.com/espressif/esp-sr/releases/download/v1.0.0/sr_model.tar.gz
tar -xzf sr_model.tar.gz -C /sdcard/sr_model
```

> **注意**：模型文件较大（约 10MB），需要 SD 卡支持。如无 SD 卡，可使用简化版模型或考虑云端方案。

## 3.2.3 步骤 2：创建语音命令模块

### 新建 `src/voice_command.h`

```cpp
// src/voice_command.h
#pragma once
#include <Arduino.h>

// 语音命令枚举
enum VoiceCommand {
    CMD_NONE = 0,
    CMD_LIGHT_ON,      // "开灯"
    CMD_LIGHT_OFF,     // "关灯"
    CMD_SHOW_TIME,     // "几点了"
    CMD_SLEEP,         // "休息一下"
};

// 初始化语音识别
void voiceCommandInit();

// 循环处理（在 loop() 中调用）
VoiceCommand voiceCommandLoop();

// 获取命令名称
const char* voiceCommandName(VoiceCommand cmd);
```

### 新建 `src/voice_command.cpp`

```cpp
// src/voice_command.cpp
#include "voice_command.h"
#include <esp_sr.h>

// 命令词模型（ESP-SR 内置）
static const char* COMMANDS[] = {
    "开灯",
    "关灯",
    "几点了",
    "休息一下",
};

static esp_sr_handle_t sr_handle = NULL;
static VoiceCommand _lastCommand = CMD_NONE;

void voiceCommandInit() {
    // 配置语音识别
    esp_sr_config_t config = ESP_SR_DEFAULT_CONFIG();
    config.model_path = "/sdcard/sr_model";  // 模型文件路径
    config.afe_mode = SR_LOW_ENERGY;         // 低功耗模式

    sr_handle = esp_sr_create(&config);

    // 添加命令词
    for (const char* cmd : COMMANDS) {
        esp_sr_add_command(sr_handle, cmd);
    }

    Serial.println("[Voice] 语音命令初始化完成");
}

VoiceCommand voiceCommandLoop() {
    // 等待识别结果
    if (esp_sr_is_done(sr_handle)) {
        char* result = esp_sr_get_result(sr_handle);

        Serial.printf("[Voice] 识别结果：%s\n", result);

        // 匹配命令
        if (strcmp(result, "开灯") == 0) {
            _lastCommand = CMD_LIGHT_ON;
        } else if (strcmp(result, "关灯") == 0) {
            _lastCommand = CMD_LIGHT_OFF;
        } else if (strcmp(result, "几点了") == 0) {
            _lastCommand = CMD_SHOW_TIME;
        } else if (strcmp(result, "休息一下") == 0) {
            _lastCommand = CMD_SLEEP;
        }
        
        free(result);  // 释放识别结果
    }

    VoiceCommand cmd = _lastCommand;
    _lastCommand = CMD_NONE;  // 消费后清空
    return cmd;
}

const char* voiceCommandName(VoiceCommand cmd) {
    static const char* names[] = {
        "无", "开灯", "关灯", "几点了", "休息一下"
    };
    return cmd < sizeof(names)/sizeof(names[0]) ? names[cmd] : "未知";
}
```

## 3.2.4 步骤 3：集成到主程序

### 编辑 `src/main.cpp`

```cpp
// main.cpp

#include "voice_command.h"
#include "cat_face.h"

// 外部函数声明（假设已存在）
extern void Set_Backlight(uint8_t percent);
extern CatFace catFace;

void setup() {
    Serial.begin(115200);
    
    // 现有初始化...
    
    // 初始化语音命令
    voiceCommandInit();
    
    Serial.println("[Main] 系统初始化完成");
}

void loop() {
    // 处理语音命令
    VoiceCommand cmd = voiceCommandLoop();
    if (cmd != CMD_NONE) {
        handleVoiceCommand(cmd);
    }

    // 现有逻辑（触摸、状态机等）...
    
    delay(10);
}

void handleVoiceCommand(VoiceCommand cmd) {
    Serial.printf("[Main] 处理语音命令：%s\n", voiceCommandName(cmd));

    switch (cmd) {
        case CMD_LIGHT_ON:
            // 开灯：100% 背光
            Set_Backlight(100);
            catFace.setExpression(CAT_HAPPY);
            Serial.println("[Voice] 背光已打开");
            break;

        case CMD_LIGHT_OFF:
            // 关灯：关闭背光
            Set_Backlight(0);
            catFace.setExpression(CAT_SLEEPY);
            Serial.println("[Voice] 背光已关闭");
            break;

        case CMD_SHOW_TIME:
            // 显示时间（需要 RTC 支持）
            showTimeOnScreen();
            break;

        case CMD_SLEEP:
            // 睡眠模式
            catFace.setExpression(CAT_SLEEPY);
            delay(2000);
            Set_Backlight(0);
            // 注意：esp_light_sleep_start() 会暂停 CPU，需要外部唤醒
            // esp_light_sleep_start();
            Serial.println("[Voice] 进入睡眠模式");
            break;

        default:
            break;
    }
}

    char timeStr[20];

    Serial.printf("[Voice] 时间：%s\n", timeStr);
    
    // 在 LVGL 屏幕上显示
    // （需要实现 UI 显示逻辑）
}
```

### 添加 RTC 时间支持（可选）

```cpp
// src/rtc_clock.h
#pragma once
#include <Arduino.h>

void rtcClockInit();
void rtcClockGetTime(char* timeStr, int maxSize);
bool rtcClockIsValid();

// src/rtc_clock.cpp
#include "rtc_clock.h"
#include <Wire.h>
#include <RTClib.h>  // Adafruit RTClib

static RTC_PCF85063A rtc;

void rtcClockInit() {
    Wire.begin();
    if (!rtc.begin()) {
        Serial.println("[RTC] 找不到 RTC 时钟");
        return;
    }

    if (rtc.lostPower()) {
        Serial.println("[RTC] 时钟未设置，使用编译时间");
        rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
    }
}

void rtcClockGetTime(char* timeStr, int maxSize) {
    DateTime now = rtc.now();
    snprintf(timeStr, maxSize, "%02d:%02d:%02d",
             now.hour(), now.minute(), now.second());
}

bool rtcClockIsValid() {
    return rtc.started();
}
```

### 问题：语音识别在嘈杂环境中可能误识别

### 解决方案 1：设置置信度阈值

```cpp
// voice_command.cpp 增强版

VoiceCommand voiceCommandLoop() {
    if (esp_sr_is_done(sr_handle)) {
        // 获取置信度
        float confidence = esp_sr_get_confidence(sr_handle);
        char* result = esp_sr_get_result(sr_handle);

        Serial.printf("[Voice] 识别结果：%s (置信度：%.2f)\n", 
                      result, confidence);

        // 只接受置信度 > 80% 的结果
        if (confidence > 0.8f) {
            // ... 匹配命令逻辑
        } else {
            Serial.println("[Voice] 置信度过低，忽略");
        }
        
        free(result);
    }
    
    // ...
}
```

### 解决方案 2：添加唤醒词

```cpp
// voice_command.cpp - 带唤醒词

static bool _isAwake = false;
static uint32_t _wakeTimeout = 0;

VoiceCommand voiceCommandLoop() {
    if (esp_sr_is_done(sr_handle)) {
        char* result = esp_sr_get_result(sr_handle);

        if (!_isAwake) {
            // 未唤醒状态，只检测唤醒词
            if (strcmp(result, "小喵小喵") == 0) {
                _isAwake = true;
                _wakeTimeout = millis() + 5000;  // 5 秒超时
                catFace.setExpression(CAT_LISTENING);
                Serial.println("[Voice] 已唤醒，请说命令");
            }
        } else {
            // 已唤醒，检测命令超时
            if (millis() > _wakeTimeout) {
                _isAwake = false;
                catFace.setExpression(CAT_NEUTRAL);
                Serial.println("[Voice] 唤醒超时");
            }
            
            // 检测命令词
            if (strcmp(result, "开灯") == 0) {
                _lastCommand = CMD_LIGHT_ON;
                _isAwake = false;  // 执行后退出唤醒状态
            }
            // ... 其他命令
        }
        
        free(result);
    }
    
    // ...
}
```

### 解决方案 3：使用 AFE（音频前端）降噪

```cpp
// voice_command.cpp - 高级配置

void voiceCommandInit() {
    esp_sr_config_t config = ESP_SR_DEFAULT_CONFIG();
    config.model_path = "/sdcard/sr_model";
    
    // 启用高性能模式（多麦克风波束成形）
    config.afe_mode = SR_HIGH_PERF;
    
    // 配置 AFE 参数
    config.afe_config = AFE_CONFIG_DEFAULT();
    config.afe_config->aec_enable = true;   // 回声消除
    config.afe_config->ns_enable = true;    // 降噪
    config.afe_config->bf_enable = true;    // 波束成形

    sr_handle = esp_sr_create(&config);

    // 添加命令词（包括唤醒词）
    esp_sr_add_command(sr_handle, "小喵小喵");  // 唤醒词
    for (const char* cmd : COMMANDS) {
        esp_sr_add_command(sr_handle, cmd);
    }

    Serial.println("[Voice] 语音命令初始化完成（高性能模式）");
}
```

### 解决方案 4：多轮确认（关键命令）

```cpp
// 对于关键命令，要求二次确认

enum ConfirmState {
    CONFIRM_IDLE,
    CONFIRM_WAITING,
};

static ConfirmState _confirmState = CONFIRM_IDLE;
static VoiceCommand _pendingCommand = CMD_NONE;

VoiceCommand voiceCommandLoop() {
    if (esp_sr_is_done(sr_handle)) {
        char* result = esp_sr_get_result(sr_handle);

        if (_confirmState == CONFIRM_WAITING) {
            // 等待确认
            if (strcmp(result, "确认") == 0 || strcmp(result, "是的") == 0) {
                Serial.println("[Voice] 确认执行");
                VoiceCommand cmd = _pendingCommand;
                _confirmState = CONFIRM_IDLE;
                _pendingCommand = CMD_NONE;
                free(result);
                return cmd;
            } else if (strcmp(result, "取消") == 0) {
                Serial.println("[Voice] 已取消");
                _confirmState = CONFIRM_IDLE;
                _pendingCommand = CMD_NONE;
            }
        } else {
            // 正常识别
            // ... 匹配命令逻辑
            
            // 对于关键命令，进入确认状态
            if (cmd == CMD_SLEEP) {
                _confirmState = CONFIRM_WAITING;
                _pendingCommand = cmd;
                Serial.println("[Voice] 请说'确认'执行，或'取消'放弃");
                catFace.setExpression(CAT_THINKING);
            }
        }
        
        free(result);
    }
    
    // ...
}
```

### 常见问题排查

| 问题 | 可能原因 | 解决方法 |
|------|---------|---------|
| 识别率低 | 环境噪音大 | 启用 AFE 降噪，提高置信度阈值 |
| 无反应 | 麦克风增益低 | 调整 ES7210 增益寄存器 |
| 模型加载失败 | SD 卡未挂载 | 检查 SD 卡连接，确认模型路径 |
| 内存不足 | PSRAM 不足 | 减小缓冲区，使用简化模型 |

## 常见问题

**Q: ESP-SR 库编译失败？**  
A: 检查 platformio.ini 中的 `build_flags`，确保 `CONFIG_ESP_SR_LIB_ENABLE` 已定义。

**Q: 模型文件放在哪里？**  
A: 默认路径 `/sdcard/sr_model`，需要 SD 卡支持。也可修改为 SPIFFS 路径。

**Q: 识别响应慢？**  
A: 检查 AFE 模式，`SR_LOW_ENERGY` 比 `SR_HIGH_PERF` 快但准确率略低。
