```cpp
// 初始化 WiFi
void wifiManagerInit(const char* ssid, const char* password);

// 检查连接状态
bool wifiManagerIsConnected();

// 注册状态回调
void wifiManagerOnStatusChange(WiFiStatusCb cb);

// 获取 IP 地址
String wifiManagerGetIP();

// 循环处理（在 loop() 中调用）
void wifiManagerLoop();
```

### 新建 `src/wifi_manager.cpp`

```cpp
// src/wifi_manager.cpp
#include "wifi_manager.h"
#include <WiFi.h>

static WiFiStatusCb _statusCb = nullptr;
static bool _wasConnected = false;
static uint32_t _reconnectDelay = 5000;  // 5 秒重连
static uint32_t _lastReconnectMs = 0;
static bool _isReconnecting = false;

void wifiManagerInit(const char* ssid, const char* password) {
    Serial.print("[WiFi] 正在连接 ");
    Serial.println(ssid);
    
    WiFi.begin(ssid, password);
    
    // 等待初次连接
    int timeout = 10000;  // 10 秒超时
    while (WiFi.status() != WL_CONNECTED && timeout > 0) {
        delay(100);
        Serial.print(".");
        timeout -= 100;
    }
    
    if (WiFi.status() == WL_CONNECTED) {
        Serial.println("\n[WiFi] 已连接");
        Serial.printf("[WiFi] IP: %s\n", WiFi.localIP().toString().c_str());
        _wasConnected = true;
    } else {
        Serial.println("\n[WiFi] 连接失败，进入离线模式");
        _wasConnected = false;
    }
}

bool wifiManagerIsConnected() {
    return WiFi.status() == WL_CONNECTED;
}

void wifiManagerOnStatusChange(WiFiStatusCb cb) {
    _statusCb = cb;
}

String wifiManagerGetIP() {
    if (wifiManagerIsConnected()) {
        return WiFi.localIP().toString();
    }
    return "N/A";
}

void wifiManagerLoop() {
    bool connected = wifiManagerIsConnected();
    
    // 状态变化时通知回调
    if (connected != _wasConnected && _statusCb != nullptr) {
        _statusCb(connected);
    }
    _wasConnected = connected;
    
    // 断开时自动重连
    if (!connected && millis() - _lastReconnectMs > _reconnectDelay) {
        Serial.println("[WiFi] 尝试重连...");
        
        if (_isReconnecting) {
            WiFi.reconnect();
        } else {
            WiFi.begin(WiFi.getSSID().c_str(), WiFi.getPSK().c_str());
            _isReconnecting = true;
        }
        
        _lastReconnectMs = millis();
    }
}
```

### 本地时间显示

```cpp
// src/rtc_clock.h
#pragma once
#include <Arduino.h>

void rtcClockInit();
void rtcClockGetTime(char* timeStr, int maxSize);
bool rtcClockIsValid();

// src/rtc_clock.cpp
#include "rtc_clock.h"
#include <Wire.h>
#include <RTClib.h>  // Adafruit RTClib

static RTC_PCF85063A rtc;

void rtcClockInit() {
    Wire.begin();
    if (!rtc.begin()) {
        Serial.println("[RTC] 找不到 RTC 时钟");
        return;
    }

    if (rtc.lostPower()) {
        Serial.println("[RTC] 时钟未设置，使用编译时间");
        rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
    }
    
    Serial.println("[RTC] 初始化完成");
}

void rtcClockGetTime(char* timeStr, int maxSize) {
    DateTime now = rtc.now();
    snprintf(timeStr, maxSize, "%02d:%02d:%02d",
             now.hour(), now.minute(), now.second());
}

bool rtcClockIsValid() {
    return rtc.started();
}
```

```cpp
// main.cpp - 离线模式触摸处理

void handleLocalTouch() {
    Gesture gesture = getTouchGesture();
    
    switch (gesture) {
        case SINGLE_CLICK:
            // 单击：开心表情
            catFace.setExpression(CAT_HAPPY);
            Serial.println("[Offline] 单击 → 开心");
            break;
            
        case LONG_PRESS:
            // 长按：进入睡眠
            catFace.setExpression(CAT_SLEEPY);
            delay(1000);
            Set_Backlight(0);
            Serial.println("[Offline] 长按 → 睡眠");
            break;
            
        case SWIPE_UP:
            // 上滑：显示时间
            if (rtcClockIsValid()) {
                char timeStr[20];
                rtcClockGetTime(timeStr, sizeof(timeStr));
                Serial.printf("[Offline] 时间：%s\n", timeStr);
                showTimeOnScreen(timeStr);
            } else {
                Serial.println("[Offline] RTC 未初始化");
            }
            break;
            
        case SWIPE_DOWN:
            // 下滑：切换表情
            cycleExpression();
            break;
            
        default:
            break;
    }
}

// 循环切换表情（离线模式专属）
static int _expressionIndex = 0;
void cycleExpression() {
    const CatExpression expressions[] = {
        CAT_NEUTRAL, CAT_HAPPY, CAT_THINKING, CAT_SLEEPY
    };
    const int count = sizeof(expressions) / sizeof(expressions[0]);
    
    _expressionIndex = (_expressionIndex + 1) % count;
    catFace.setExpression(expressions[_expressionIndex]);
    
    Serial.printf("[Offline] 切换表情：%d\n", _expressionIndex);
}
```

```cpp
// main.cpp

static uint32_t _lastInteractionMs = 0;
static const uint32_t IDLE_TIMEOUT_MS = 120000;  // 2 分钟无操作

void recordInteraction() {
    _lastInteractionMs = millis();
    
    // 从睡眠唤醒
    if (_systemMode == MODE_OFFLINE && getBacklight() == 0) {
        Set_Backlight(50);
        catFace.setExpression(CAT_NEUTRAL);
    }
}

void checkIdleTimeout() {
    if (_systemMode != MODE_ONLINE) return;  // 离线模式不自动睡眠
    
    if (millis() - _lastInteractionMs > IDLE_TIMEOUT_MS) {
        catFace.setExpression(CAT_SLEEPY);
        Serial.println("[Main] 待机超时，进入睡眠");
    }
}

    // ... 触摸处理 ...

}
```

## 3.4.5 步骤 4：重连后的同步

### 离线期间的交互记录

```cpp
// src/offline_queue.h
#pragma once
#include <Arduino.h>

#define OFFLINE_QUEUE_SIZE 10

struct OfflineInteraction {
    uint32_t timestamp;
    char transcript[256];
    char response[256];
};

class OfflineQueue {
private:
    OfflineInteraction _queue[OFFLINE_QUEUE_SIZE];
    int _count;
    int _head;
    int _tail;
    
public:
    OfflineQueue() : _count(0), _head(0), _tail(0) {}
    
    // 添加交互记录
    bool enqueue(const char* transcript, const char* response);
    
    // 获取并删除第一条记录
    bool dequeue(OfflineInteraction* item);
    
    // 获取队列中的记录数
    int count() const { return _count; }
    
    // 清空队列
    void clear();
};

// src/offline_queue.cpp
#include "offline_queue.h"
#include <string.h>

bool OfflineQueue::enqueue(const char* transcript, const char* response) {
    if (_count >= OFFLINE_QUEUE_SIZE) {
        // 队列满，丢弃最旧的
        _tail = (_tail + 1) % OFFLINE_QUEUE_SIZE;
        _count--;
    }
    
    OfflineInteraction* item = &_queue[_head];
    item->timestamp = millis();
    strncpy(item->transcript, transcript, 255);
    strncpy(item->response, response, 255);
    
    _head = (_head + 1) % OFFLINE_QUEUE_SIZE;
    _count++;
    
    return true;
}

bool OfflineQueue::dequeue(OfflineInteraction* item) {
    if (_count == 0) return false;
    
    *item = _queue[_tail];
    _tail = (_tail + 1) % OFFLINE_QUEUE_SIZE;
    _count--;
    
    return true;
}

void OfflineQueue::clear() {
    _count = 0;
    _head = 0;
    _tail = 0;
}
```

### 重连后同步数据

```cpp
// main.cpp

#include "offline_queue.h"

static OfflineQueue _offlineQueue;

void syncOfflineInteractions() {
    if (_offlineQueue.count() == 0) return;
    
    Serial.printf("[Sync] 同步 %d 条离线记录\n", _offlineQueue.count());
    
    // 逐条同步到后端
    OfflineInteraction item;
    while (_offlineQueue.dequeue(&item)) {
        // 这里可以调用后端 API 同步
        // 由于后端可能已有变更，简单处理是只记录日志
        Serial.printf("[Sync] 转录：%s\n", item.transcript);
        Serial.printf("[Sync] 回复：%s\n", item.response);
    }
    
    Serial.println("[Sync] 同步完成");
}

// WiFi 状态变化回调
void onWiFiStatusChange(bool connected) {
    if (connected && _systemMode == MODE_OFFLINE) {
        // 从离线切换到在线，同步数据
        Serial.println("[WiFi] 重连成功，准备同步");
        delay(1000);  // 等待网络稳定
        syncOfflineInteractions();
        
        _systemMode = MODE_ONLINE;
        catFace.setExpression(CAT_HAPPY);
    } else if (!connected && _systemMode == MODE_ONLINE) {
        _systemMode = MODE_OFFLINE;
        catFace.setExpression(CAT_THINKING);
    }
}
```

## 常见问题

**Q: 重连后无法同步数据？**  
A: 检查后端 API 是否接受延迟同步的数据，可能需要特殊的同步接口。

**Q: 离线模式耗电快？**  
A: 启用 ESP32 的轻睡眠模式，定时唤醒检测触摸。

**Q: RTC 时间不准？**  
A: 在线时定期用 NTP 校准 RTC，或改用带晶体的 RTC 模块。
