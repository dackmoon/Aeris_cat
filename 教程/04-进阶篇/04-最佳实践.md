```cpp
// src/logger.h
#pragma once
#include <Arduino.h>

// 日志级别
enum LogLevel {
    LOG_DEBUG = 0,
    LOG_INFO,
    LOG_WARNING,
    LOG_ERROR,
    LOG_NONE  // 禁用
};

// 日志模块
enum LogModule {
    MOD_MAIN = 0,
    MOD_WIFI,
    MOD_AUDIO,
    MOD_DISPLAY,
    MOD_CLOUD,
    MOD_OTA,
};

class Logger {
private:
    LogLevel _level;
    static const size_t BUFFER_SIZE = 256;
    
public:
    void init(LogLevel level = LOG_INFO) {
        _level = level;
        Serial.begin(115200);
    }
    
    void setLevel(LogLevel level) {
        _level = level;
    }
    
    // 格式化日志输出
    void log(LogLevel level, LogModule module, const char* format, ...) {
        if (level < _level) return;
        
        static const char* levelNames[] = {"DBG", "INF", "WRN", "ERR"};
        static const char* moduleNames[] = {"MAIN", "WIFI", "AUDIO", "DISP", "CLOUD", "OTA"};
        
        char buffer[BUFFER_SIZE];
        
        // 时间戳
        uint32_t ms = millis();
        int len = snprintf(buffer, BUFFER_SIZE, "[%lu] [%s] [%s] ", 
                           ms, levelNames[level], moduleNames[module]);
        
        // 可变参数格式化
        va_list args;
        va_start(args, format);
        vsnprintf(buffer + len, BUFFER_SIZE - len, format, args);
        va_end(args);
        
        Serial.println(buffer);
        
        // 错误日志额外打印堆栈
        if (level == LOG_ERROR) {
            printStackInfo();
        }
    }
    
    void printStackInfo() {
        // ESP32-S3 堆栈信息
        Serial.printf("Free stack: %d\n", uxTaskGetStackHighWaterMark(NULL));
    }
    
// 宏定义简化调用
#define LOG_D(module, fmt, ...) \
    logger.log(LOG_DEBUG, module, fmt, ##__VA_ARGS__)
#define LOG_I(module, fmt, ...) \
    logger.log(LOG_INFO, module, fmt, ##__VA_ARGS__)
#define LOG_W(module, fmt, ...) \
    logger.log(LOG_WARNING, module, fmt, ##__VA_ARGS__)
#define LOG_E(module, fmt, ...) \
    logger.log(LOG_ERROR, module, fmt, ##__VA_ARGS__)
};

// 全局日志实例
extern Logger logger;
```

# utils/logging_config.py
import logging
from logging.handlers import RotatingFileHandler
import sys
from datetime import datetime

def setup_logging(log_level: str = "INFO", log_file: str = "/var/log/cybercat/backend.log"):
    """配置日志系统"""
    
    # 创建日志格式
    formatter = logging.Formatter(
        fmt='%(asctime)s.%(msecs)03d | %(levelname)-8s | %(name)s:%(lineno)d | %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S'
    )
    
    # 控制台处理器
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setFormatter(formatter)
    
    # 文件处理器（轮转，最大 10MB，保留 5 个文件）
    file_handler = RotatingFileHandler(
        log_file,
        maxBytes=10 * 1024 * 1024,
        backupCount=5,
        encoding='utf-8'
    )
    file_handler.setFormatter(formatter)
    
    # 根日志器
    root_logger = logging.getLogger()
    root_logger.setLevel(log_level)
    root_logger.addHandler(console_handler)
    root_logger.addHandler(file_handler)
    
    # 静音某些库的日志
    logging.getLogger("uvicorn.access").setLevel(logging.WARNING)
    logging.getLogger("httpx").setLevel(logging.WARNING)
    
    logging.info("日志系统初始化完成")

# middleware/metrics.py
from prometheus_client import Counter, Histogram, generate_latest
from fastapi import Response
import time

# 定义指标
REQUEST_COUNT = Counter(
    'cybercat_request_total',
    'Total requests',
    ['endpoint', 'method', 'status']
)

REQUEST_DURATION = Histogram(
    'cybercat_request_duration_seconds',
    'Request duration',
    ['endpoint']
)

ACTIVE_CONNECTIONS = Counter(
    'cybercat_active_connections',
    'Active database connections',
    ['state']  # 'idle', 'active'
)

async def metrics_middleware(request, call_next):
    """Prometheus 指标采集中间件"""
    start_time = time.time()
    
    try:
        response = await call_next(request)
        
        # 记录指标
        REQUEST_COUNT.labels(
            endpoint=request.url.path,
            method=request.method,
            status=response.status_code
        ).inc()
        
        duration = time.time() - start_time
        REQUEST_DURATION.labels(endpoint=request.url.path).observe(duration)
        
        return response
    except Exception as e:
        REQUEST_COUNT.labels(
            endpoint=request.url.path,
            method=request.method,
            status=500
        ).inc()
        raise

# 暴露指标端点
@app.get("/metrics")

    """Prometheus 指标抓取"""
    return Response(content=generate_latest(), media_type="text/plain")

"""

提供基于规则的情绪分类功能，支持以下情绪类型：
- neutral: 中性/默认
- happy: 开心/兴奋

from typing import List, Dict

    "angry": ["哼", "讨厌", "讨厌你", "不理你", "生气了", "坏蛋"],
    "happy": ["!", "～", "~", "哈哈", "嘻嘻", "嘿嘿", "喵喵", "喵~"],
    "thinking": ["嗯", "呃", "这个", "让我想想", "也许", "可能", "大概"],

    """
    分析文本情绪（规则匹配版）

    Returns:

        ValueError: 当输入文本为空时

        >>> analyze_emotion("你在干什么吗？")
        'confused'

// src/performance_budget.h

// 定义性能预算阈值
#define BUDGET_LOOP_MAX_MS      50      // 主循环最大耗时
#define BUDGET_ANIMATION_MAX_MS 16      // 单帧动画耗时（60fps）
#define BUDGET_AUDIO_MAX_MS     10      // 音频处理耗时
#define BUDGET_HEAP_MIN_KB      100     // 最小空闲 Heap
#define BUDGET_PSRAM_MIN_KB     500     // 最小空闲 PSRAM

// 预算检查宏
#define CHECK_BUDGET(fn, budget_ms) do { \
    uint32_t start = micros(); \
    fn; \
    uint32_t elapsed = (micros() - start) / 1000; \
    if (elapsed > budget_ms) { \
        LOG_W(MOD_MAIN, "预算超支：%s 耗时 %dms > %dms", #fn, elapsed, budget_ms); \
    } \
} while(0)

// 使用示例
void loop() {
    CHECK_BUDGET(catFace.update(), BUDGET_ANIMATION_MAX_MS);
    CHECK_BUDGET(handleAudio(), BUDGET_AUDIO_MAX_MS);
    
    uint32_t loopStart = micros();
    
    // ... 其他逻辑
    
    uint32_t loopElapsed = (micros() - loopStart) / 1000;
    if (loopElapsed > BUDGET_LOOP_MAX_MS) {
        LOG_E(MOD_MAIN, "主循环严重超支：%dms", loopElapsed);
    }
}
